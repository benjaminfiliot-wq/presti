function copier(source, dest, outil = false) {
  fs.mkdirSync(dest,{recursive:true})
  const fichiers = git(source,['ls-files','-z','--cached','--others','--exclude-standard']).split('\0').filter(Boolean)
  for (const f of fichiers) {
    if (!outil && !/^(app\/|lib\/|scripts\/|docs\/|\.github\/|[^/]+$)/.test(f)) continue
    if (/^(\.git\/|node_modules\/|\.next\/|\.env)/.test(f)) continue
    const de=path.join(source,f), vers=path.join(dest,f)
    if (!fs.existsSync(de) || !fs.lstatSync(de).isFile()) continue
    fs.mkdirSync(path.dirname(vers),{recursive:true}); fs.copyFileSync(de,vers)
  }
  if (!outil) {
    for (const f of ['next-env.d.ts','tsconfig.json']) if(fs.existsSync(path.join(source,f))) fs.copyFileSync(path.join(source,f),path.join(dest,f))
    const lien=path.join(dest,'node_modules'); fs.symlinkSync(path.join(source,'node_modules'),lien,process.platform==='win32'?'junction':'dir'); jonctions.push(lien)
  }
}

  if(g==='media') a.push(`--git=${ROOT}`)
  if(g==='next') a.push('--statique')
  return a
}

  const sortie=(x.stdout||'')+(x.stderr||'')
  return {code:x.status??124,sortie,duree_s:(Date.now()-debut)/1000}
}
