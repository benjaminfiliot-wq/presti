const P = new URLSearchParams(location.search);
const FMT = P.get('f') || 'v';                       // v | c | p
const DIM = {v:[1080,1920], c:[1080,1350], p:[1920,1080]}[FMT];
const W = DIM[0], H = DIM[1];
document.documentElement.style.setProperty('--W', W+'px');
document.documentElement.style.setProperty('--H', H+'px');
const CFG={v:{K:1,cw:1,cy:1},c:{K:.88,cw:.74,cy:.90},p:{K:.62,cw:.30,cy:1}}[FMT];
const K = (W/1080)*CFG.K;   // échelle typo
const CW = CFG.cw, CY = CFG.cy;   // échelle et centre des cartes
const DUR = 22;

// ---------- utilitaires ----------
const clamp=(x,a=0,b=1)=>Math.max(a,Math.min(b,x));
const eOut=x=>1-Math.pow(1-x,3);
const eExpo=x=>x>=1?1:1-Math.pow(2,-9*x);
const eInOut=x=>x<.5?4*x*x*x:1-Math.pow(-2*x+2,3)/2;
// progression locale d'un plan
const seg=(t,a,b)=>clamp((t-a)/(b-a));
const live=(t,a,b)=>t>=a-.001&&t<b;

function setShot(el,on,op=1){ el.style.opacity = on?op:0; }

// mots animés : chaque mot entre avec décalage
function words(el, txt, t, t0, per=.075, dur=.5, accent=null, size=1, gap='.28em'){
  if(el.dataset.txt!==txt){
    el.dataset.txt=txt; el.innerHTML='';
    txt.split('\n').forEach((line,li)=>{
      const d=document.createElement('div'); d.style.display='block';
      line.split(' ').forEach((w,i)=>{
        const s=document.createElement('span'); s.className='word'; s.textContent=w;
        s.style.marginRight=gap;
        if(accent && accent.includes(w.replace(/[.,]/g,''))) s.style.color='#E8946F';
        d.appendChild(s);
      });
      el.appendChild(d);
    });
  }
  el.style.fontSize=(size*K)+'px';
  const ws=el.querySelectorAll('.word');
  ws.forEach((s,i)=>{
    const p=eExpo(seg(t, t0+i*per, t0+i*per+dur));
    s.style.opacity=p;
    s.style.transform=`translate3d(0,${(1-p)*0.34*size*K}px,0) scale(${0.94+0.06*p})`;
  });
}

// place une carte : largeur en fraction de W, centre vertical en fraction de H
function place(el,fw,cy,scale=1){const w=W*fw*CW,h=w/AR[el.id];el.style.width=w+'px';el.style.left=((W-w)/2)+'px';el.style.top=(H*cy*CY-h/2)+'px';return h;}
for(const k in IMG) document.getElementById(k).src=IMG[k];
