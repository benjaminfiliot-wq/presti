// px du monde ; y vers le bas, z vers l'œil. Chêne : trois tons.
const pieces = []
const f = (cls, w, h, tr, extra = '') => `<i class="f ${cls}" style="width:${w}px;height:${h}px;margin:${-h / 2}px 0 0 ${-w / 2}px;transform:${tr}">${extra}</i>`
function bloc({ nom, w, h, d, x, y, z, ex, ey, ez, fil = 'x', etiq, lx = 0, ly = 0, lz = 0, decor = {} }) {
  const faces = [
    f(`f--haut fil-${fil === 'x' ? 'h' : 'v'}`, w, d, `rotateX(90deg) translateZ(${h / 2}px)`, decor.haut || ''),
    f(`f--avant ${fil === 'z' ? 'bout' : 'fil-h'}`, w, h, `translateZ(${d / 2}px)`, decor.avant || ''),
    f(`f--droite ${fil === 'x' ? 'bout' : 'fil-h'}`, d, h, `rotateY(90deg) translateZ(${w / 2}px)`, decor.droite || ''),
    f('f--gauche fil-h', d, h, `rotateY(-90deg) translateZ(${w / 2}px)`),
    f('f--arriere fil-h', w, h, `rotateY(180deg) translateZ(${d / 2}px)`),
    f('f--bas fil-h', w, d, `rotateX(-90deg) translateZ(${h / 2}px)`),
  ].join('')
  const e = etiq ? `<span class="etiq" style="transform:translate3d(${lx}px,${ly}px,${lz}px) rotateY(36deg) rotateX(22deg)">${etiq}</span>` : ''
  pieces.push(`<div class="piece" data-nom="${nom}" style="--x:${x}px;--y:${y}px;--z:${z}px;--ex:${ex}px;--ey:${ey}px;--ez:${ez}px">${faces}${e}</div>`)
}

const brut = Buffer.byteLength(html), gz = zlib.gzipSync(html, { level: 9 }).length
const noeuds = (html.match(/<(i|b|span|div)\b/g) || []).length

const b = await chromium.launch({ args: ['--use-angle=d3d11', '--enable-gpu', '--ignore-gpu-blocklist', '--enable-gpu-rasterization'] })
const res = { brut, gzip: gz, elementsDuMonde: noeuds, captures: [] }
{
  const p = await b.newPage({ viewport: { width: 1440, height: 900 } })
  await p.goto('file:///' + ICI + 'proto-css.html')
  res.gpu = await p.evaluate(() => { const g = document.createElement('canvas').getContext('webgl'); const i = g && g.getExtension('WEBGL_debug_renderer_info'); return g ? String(g.getParameter(i ? i.UNMASKED_RENDERER_WEBGL : g.RENDERER)) : 'aucun' })
  for (const v of [0, 0.35, 0.7, 1]) {
    await p.evaluate((v) => document.documentElement.style.setProperty('--p', v), v)
    await p.waitForTimeout(150)
    const nom = `css-bureau-p${String(v * 100).padStart(3, '0')}.png`
    await p.screenshot({ path: ICI + nom }); res.captures.push(nom)
  }
  // balayage 0 → 1 → 0 en 3 s, 3 fois : intervalles entre images
  res.balayage = await p.evaluate(() => new Promise((ok) => {
    const d = []; let t0 = performance.now(), prev = t0
    const pas = (t) => {
      d.push(t - prev); prev = t
      const k = ((t - t0) / 3000) % 2, v = k < 1 ? k : 2 - k
      document.documentElement.style.setProperty('--p', v.toFixed(4))
      if (t - t0 < 9000) requestAnimationFrame(pas); else { d.shift(); d.sort((a, b) => a - b); ok({ images: d.length, medianeMs: +d[Math.floor(d.length / 2)].toFixed(2), p95Ms: +d[Math.floor(d.length * 0.95)].toFixed(2), maxMs: +d[d.length - 1].toFixed(2), auDelaDe25ms: d.filter((x) => x > 25).length }) }
    }
    requestAnimationFrame(pas)
  }))
  await p.close()
}
{
  const ctx = await b.newContext({ viewport: { width: 390, height: 844 }, deviceScaleFactor: 2, isMobile: true, hasTouch: true })
  const p = await ctx.newPage()
  await p.goto('file:///' + ICI + 'proto-css.html')
  for (const v of [0, 1]) {
    await p.evaluate((v) => document.documentElement.style.setProperty('--p', v), v)

    const nom = `css-tel-p${String(v * 100).padStart(3, '0')}.png`
    await p.screenshot({ path: ICI + nom }); res.captures.push(nom)
  }
  await ctx.close()
}
