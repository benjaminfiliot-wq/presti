// Outil de mise au point (pas un juge) : refait le composite des calques comme scripts/eclate-rendu-qa.mjs (OB-4) et écrit
// une carte des pixels écartés de plus de 24 niveaux, pour voir OÙ le composite diffère de eclate.avif. Mesure aussi les
// marges du cadre (OB-10) et l'alpha maximal des pixels d'ombre (noirs, alpha < 250) de chaque calque.
// Usage : node outils/ecarts.mjs [dossier contenant eclate.json et les AVIF]
import fs from 'node:fs'
import path from 'node:path'
import { createRequire } from 'node:module'
import { fileURLToPath } from 'node:url'

const require = createRequire('C:/Users/benja/Documents/presti/package.json')
const sharp = require('sharp')
const ICI = path.resolve(path.dirname(fileURLToPath(import.meta.url)), '..')
const REN = path.resolve(process.argv[2] || path.join(ICI, 'rendus'))
const L = 1400, H = 1050
const j = JSON.parse(fs.readFileSync(path.join(REN, 'eclate.json'), 'utf8'))
const verite = await sharp(path.join(REN, 'eclate.avif')).flatten({ background: '#000' }).resize(L, H).raw().toBuffer()
const fond = { r: verite[0], g: verite[1], b: verite[2] }
const poses = []
for (const p of [...j.pieces].sort((a, b) => a.ordre - b.ordre)) {
  const bw = Math.round(p.l * L), bh = Math.round(p.h * H)
  const buf = await sharp(path.join(REN, p.fichier)).resize(bw, bh, { fit: 'fill' }).png().toBuffer()
  const lx = Math.round((p.x + p.vecteur[0]) * L), ly = Math.round((p.y + p.vecteur[1]) * H)
  poses.push({ input: buf, left: lx, top: ly })
  for (const [vue, dx, dy] of [['assemblée', 0, 0], ['éclatée', p.vecteur[0], p.vecteur[1]]]) {
    const x0 = p.x + dx, y0 = p.y + dy
    console.log(`${p.nom} ${vue} : marges g ${x0.toFixed(3)} d ${(1 - x0 - p.l).toFixed(3)} h ${y0.toFixed(3)} b ${(1 - y0 - p.h).toFixed(3)}`)
  }
  const { data } = await sharp(path.join(REN, p.fichier)).ensureAlpha().raw().toBuffer({ resolveWithObject: true })
  let aMax = 0, n = 0
  for (let i = 0; i < data.length; i += 4) if (data[i + 3] < 250 && Math.max(data[i], data[i + 1], data[i + 2]) < 12) { aMax = Math.max(aMax, data[i + 3]); n++ }
  console.log(`${p.nom} : alpha max des pixels d'ombre ${(aMax / 255).toFixed(3)} (${n} pixels)`)
}
const comp = await sharp({ create: { width: L, height: H, channels: 3, background: fond } }).composite(poses).removeAlpha().raw().toBuffer()
const carte = Buffer.alloc(L * H * 3)
let n = 0
for (let i = 0, k = 0; i < comp.length; i += 3, k++) {
  const d = Math.max(Math.abs(comp[i] - verite[i]), Math.abs(comp[i + 1] - verite[i + 1]), Math.abs(comp[i + 2] - verite[i + 2]))
  const g = verite[i + 1] >> 2
  if (d > 24) { n++; carte[i] = 255; carte[i + 1] = 0; carte[i + 2] = 0 } else { carte[i] = g; carte[i + 1] = g; carte[i + 2] = g }
}
console.log(`écartés : ${n} pixels, ${(100 * n / (L * H)).toFixed(2)} %`)
fs.mkdirSync(path.join(ICI, 'regard'), { recursive: true })
await sharp(carte, { raw: { width: L, height: H, channels: 3 } }).png().toFile(path.join(ICI, 'regard', 'carte-ecarts.png'))
