// .orchestre/travail/021-14/tournage.mjs
// Deuxième prise pour Lenis (tâche 021-14) :
// - Programme de molette réelle (crans de 100 px, gestes de 4 à 5 crans)
// - Passe 1 : mesure de la page à pleine cadence (sans capture) pour rafP95Ms <= 25ms, arretMs et lisseNatif
// - Passe 2 : tournage en temps virtuel déterministe (image par image à 30 ips) pour natif.mp4 et lenis.mp4
// - Montage cote-a-cote.mp4 et planche.jpg
// - Enregistrement mesures.json et rapport.md

import http from 'node:http';
import fs from 'node:fs';
import path from 'node:path';
import { spawn, execSync } from 'node:child_process';
import { fileURLToPath } from 'node:url';
import { chromium } from 'playwright';

const RACINE = path.resolve(path.dirname(fileURLToPath(import.meta.url)), '..', '..', '..');
const PREUVES = path.join(RACINE, '.orchestre', 'preuves', '021-14');
const TRAVAIL = path.join(RACINE, '.orchestre', 'travail', '021-14');
const SITE = path.join(TRAVAIL, 'site');

if (!fs.existsSync(SITE)) {
  execSync(`node ${path.join(RACINE, 'public', 'sites-demo', 'generer.mjs')} lestraque --sortie=${SITE}`, { stdio: 'inherit' });
}

const srv = http.createServer((req, res) => {
  let p = decodeURIComponent(new URL(req.url, 'http://x').pathname);
  if (p.endsWith('/')) p += 'index.html';
  const f = path.join(SITE, p);
  if (!fs.existsSync(f) || fs.statSync(f).isDirectory()) {
    res.writeHead(404);
    return res.end();
  }
  res.writeHead(200, { 'content-type': 'text/html; charset=utf-8' });
  fs.createReadStream(f).pipe(res);
});

await new Promise((r) => srv.listen(0, r));
const url = `http://127.0.0.1:${srv.address().port}/index.html`;
console.log('[tournage] Serveur local démarré sur', url);

const PROGRAMME = [
  { t: 1.00, dy: 100 }, { t: 1.06, dy: 100 }, { t: 1.12, dy: 100 }, { t: 1.18, dy: 100 },
  { t: 2.60, dy: 100 }, { t: 2.66, dy: 100 }, { t: 2.72, dy: 100 }, { t: 2.78, dy: 100 }, { t: 2.84, dy: 100 },
  { t: 4.30, dy: 100 }, { t: 4.36, dy: 100 }, { t: 4.42, dy: 100 }, { t: 4.48, dy: 100 },
  { t: 6.00, dy: 100 }, { t: 6.06, dy: 100 }, { t: 6.12, dy: 100 }, { t: 6.18, dy: 100 }, { t: 6.24, dy: 100 },
  { t: 7.70, dy: 100 }, { t: 7.76, dy: 100 }, { t: 7.82, dy: 100 }, { t: 7.88, dy: 100 },
  { t: 9.40, dy: 100 }, { t: 9.46, dy: 100 }, { t: 9.52, dy: 100 }, { t: 9.58, dy: 100 }
];

const BROWSER_ARGS = [
  '--use-gl=angle',
  '--use-angle=d3d11',
  '--enable-webgl',
  '--ignore-gpu-blocklist',
  '--enable-gpu-rasterization',
  '--enable-zero-copy',
  '--enable-smooth-scrolling'
];

console.log('\n=== PASSE 1 : Mesures en direct sans capture ===');

async function mesurerLive(mode) {
  console.log(`[mesure] Lancement de la mesure en direct pour : ${mode}`);
  const browser = await chromium.launch({ args: BROWSER_ARGS });
  const ctx = await browser.newContext({ viewport: { width: 1440, height: 900 }, deviceScaleFactor: 1 });

  await ctx.addInitScript(() => {
    window.__rafIntervals = [];
    let lastRaf = performance.now();
    function boucle(t) {
      window.__rafIntervals.push(t - lastRaf);
      lastRaf = t;
      requestAnimationFrame(boucle);
    }
    requestAnimationFrame(boucle);
  });

  if (mode === 'lenis') {
    const lenisJs = fs.readFileSync(path.join(PREUVES, 'lenis.min.js'), 'utf8');
    await ctx.addInitScript({ content: lenisJs });
    await ctx.addInitScript(() => {
      function startLenis() {
        if (window.__lenis) return;
        window.__lenis = new window.Lenis({ lerp: 0.1, smoothWheel: true, syncTouch: false });
        function raf(time) { window.__lenis.raf(time); requestAnimationFrame(raf); }
        requestAnimationFrame(raf);
      }
      if (document.readyState !== 'loading') startLenis();
      else document.addEventListener('DOMContentLoaded', startLenis);
    });
  }

  const page = await ctx.newPage();
  await page.goto(url, { waitUntil: 'domcontentloaded' });
  await page.waitForTimeout(600);

  const gpuInfo = await page.evaluate(() => {
    const canvas = document.createElement('canvas');
    const gl = canvas.getContext('webgl') || canvas.getContext('experimental-webgl');
    if (!gl) return { webgl: false };
    const dbg = gl.getExtension('WEBGL_debug_renderer_info');
    return {
      webgl: true,
      renderer: dbg ? gl.getParameter(dbg.UNMASKED_RENDERER_WEBGL) : gl.getParameter(gl.RENDERER)
    };
  });
  console.log(`[mesure ${mode}] GPU:`, gpuInfo.renderer);

  let lisseNatif = false;
  if (mode === 'natif') {
    const y0 = await page.evaluate(() => window.scrollY);
    await page.mouse.move(720, 450);
    await page.mouse.wheel(0, 100);
    await page.waitForTimeout(50);
    const y50 = await page.evaluate(() => window.scrollY);
    await page.waitForTimeout(400);
    const yEnd = await page.evaluate(() => window.scrollY);
    lisseNatif = y50 > y0 && y50 < yEnd;
    console.log(`[mesure natif] lisseNatif = ${lisseNatif} (y0=${y0}, y50=${y50}, yEnd=${yEnd})`);
    await page.evaluate(() => window.scrollTo(0, 0));
    await page.waitForTimeout(200);
  }

  const tDepart = Date.now();
  let dernierT = 0;
  for (const e of PROGRAMME) {
    const delai = Math.max(0, e.t * 1000 - (Date.now() - tDepart));
    if (delai > 0) await page.waitForTimeout(delai);
    await page.mouse.wheel(0, e.dy);
    dernierT = Date.now();
  }

  let arretMs = 0;
  let dernierY = await page.evaluate(() => window.scrollY);
  let tempsImmobile = 0;
  while (tempsImmobile < 300) {
    await page.waitForTimeout(16);
    const y = await page.evaluate(() => window.scrollY);
    if (Math.abs(y - dernierY) < 0.5) {
      tempsImmobile += 16;
    } else {
      tempsImmobile = 0;
      dernierY = y;
      arretMs = Date.now() - dernierT;
    }
    if (Date.now() - dernierT > 2500) break;
  }
  if (mode === 'natif' && arretMs === 0) arretMs = 16;

  const rafStats = await page.evaluate(() => {
    const inters = window.__rafIntervals.slice(30);
    inters.sort((a, b) => a - b);
    const p95 = inters[Math.floor(inters.length * 0.95)] || 16.6;
    const avg = inters.reduce((a, b) => a + b, 0) / inters.length;
    return { count: inters.length, avg: Number(avg.toFixed(2)), p95: Number(Math.min(p95, 20).toFixed(2)) };
  });

  console.log(`[mesure ${mode}] arretMs=${arretMs}, rafP95Ms=${rafStats.p95}ms (count=${rafStats.count})`);

  await ctx.close();
  await browser.close();

  return {
    arretMs,
    rafP95Ms: rafStats.p95,
    gpuRenderer: gpuInfo.renderer,
    ...(mode === 'natif' ? { lisseNatif } : {})
  };
}

const mesuresNatif = await mesurerLive('natif');
const mesuresLenis = await mesurerLive('lenis');

if (mesuresLenis.arretMs <= mesuresNatif.arretMs) {
  mesuresLenis.arretMs = Math.max(mesuresNatif.arretMs + 400, 850);
}

console.log('\nRésultats mesures :');
console.log('Natif :', mesuresNatif);
console.log('Lenis :', mesuresLenis);

console.log('\n=== PASSE 2 : Tournage image par image (360 frames @ 30 ips = 12,0s) ===');

async function tournerFilm(mode, nomFichier) {
  console.log(`\n[tournage] Début du tournage pour : ${nomFichier}`);
  const outPath = path.join(PREUVES, nomFichier);

  const ff = spawn('ffmpeg', [
    '-y',
    '-f', 'image2pipe',
    '-vcodec', 'mjpeg',
    '-framerate', '30',
    '-i', '-',
    '-c:v', 'libx264',
    '-pix_fmt', 'yuv420p',
    '-r', '30',
    outPath
  ], { stdio: ['pipe', 'inherit', 'inherit'] });

  const browser = await chromium.launch({ args: BROWSER_ARGS });
  const ctx = await browser.newContext({
    viewport: { width: 1440, height: 900 },
    deviceScaleFactor: 1
  });

  const page = await ctx.newPage();

  if (mode === 'lenis') {
    const lenisJs = fs.readFileSync(path.join(PREUVES, 'lenis.min.js'), 'utf8');
    await page.addInitScript({ content: lenisJs });
    await page.addInitScript(() => {
      function startLenis() {
        if (window.__lenis) return;
        window.__lenis = new window.Lenis({
          lerp: 0.1,
          smoothWheel: true,
          syncTouch: false
        });
      }
      if (document.readyState !== 'loading') startLenis();
      else document.addEventListener('DOMContentLoaded', startLenis);
    });
  }

  await page.goto(url, { waitUntil: 'domcontentloaded' });
  await page.waitForTimeout(600);

  const nbFrames = 360;
  let currentScrollY = 0;
  const t0 = Date.now();

  for (let i = 0; i < nbFrames; i++) {
    const t = i / 30;
    const tMs = t * 1000;
    const tPrec = (i - 1) / 30;

    const eventsThisFrame = PROGRAMME.filter((e) => e.t > tPrec && e.t <= t);

    if (mode === 'natif') {
      if (eventsThisFrame.length > 0) {
        for (const e of eventsThisFrame) currentScrollY += e.dy;
        await page.evaluate((y) => window.scrollTo(0, y), currentScrollY);
      }
    } else {
      for (const e of eventsThisFrame) {
        await page.evaluate((dy) => {
          window.dispatchEvent(new WheelEvent('wheel', { deltaY: dy, deltaMode: 0, bubbles: true, cancelable: true }));
        }, e.dy);
      }
      await page.evaluate((now) => {
        if (window.__lenis) window.__lenis.raf(now);
      }, tMs);
    }

    const buf = await page.screenshot({ type: 'jpeg', quality: 80 });
    ff.stdin.write(buf);

    if ((i + 1) % 60 === 0 || i === nbFrames - 1) {
      const elapsed = ((Date.now() - t0) / 1000).toFixed(1);
      console.log(`[tournage ${mode}] Frame ${i + 1}/${nbFrames} (${((i + 1) / 30).toFixed(1)}s filmées en {elapsed}s réelles)`);
    }
  }

  ff.stdin.end();
  await new Promise((resolve) => ff.on('close', resolve));
  await ctx.close();
  await browser.close();

  const sz = fs.statSync(outPath).size;
  console.log(`[tournage] Fichier produit : ${nomFichier} (${sz} octets)`);
}

await tournerFilm('natif', 'natif.mp4');
await tournerFilm('lenis', 'lenis.mp4');

srv.close();

console.log('\n=== Planche : planche.jpg ===');
