async function cadenceVide(browser) {
  const p = await browser.newPage()
  const ips = await p.evaluate(() => new Promise((ok) => { let n = 0; const t0 = performance.now(); const f = (t) => { n++; if (t - t0 < 2000) requestAnimationFrame(f); else ok(Math.round((n / ((t - t0) / 1000)) * 10) / 10) }; requestAnimationFrame(f) }))
  await p.close()
  return ips
}
async function attendreCalme(pourquoi, browser) {
  const debut = Date.now()
  for (let k = 0; ; k++) {
    let n = 0
    try {
      n = Number(execFileSync('powershell.exe', ['-NoProfile', '-Command', "(Get-CimInstance Win32_Process -Filter \"Name='node.exe'\" | Where-Object { $_.CommandLine -match '-qa\\.mjs' } | Measure-Object).Count"], { encoding: 'utf8' }).trim())

  }
}

// ── copies iPhone : la copie générée, recopiée, @supports réécrits ────────────────────────────────────────────────────────────
const MOTIF_SUPPORTS = /@supports\s*\(\s*animation-timeline\s*:\s*(view|scroll)\(\)\s*\)/g
function preparerCopie(nom, remplacement) {
  const src = path.join(TRAVAIL, 'www')
  const dst = path.join(TRAVAIL, `www-${nom}`)
  fs.rmSync(dst, { recursive: true, force: true })
  fs.cpSync(src, dst, { recursive: true })
  const bilan = {}
  for (const f of ['styles.css', 'bureau.css']) {
    const p = path.join(dst, 'sites-demo/lestraque', f)
    const avant = fs.readFileSync(p, 'utf8')
    const n = (avant.match(MOTIF_SUPPORTS) || []).length
    const apres = avant.replace(MOTIF_SUPPORTS, remplacement)
    fs.writeFileSync(p, apres)
    bilan[f] = { remplaces: n, restantsView: (apres.match(/animation-timeline:\s*view\(\)\s*\)/g) || []).length, restantsScroll: (apres.match(/@supports\s*\(\s*animation-timeline:\s*scroll\(\)/g) || []).length, nouveaux: apres.split(remplacement).length - 1 }
  }
  return { racine: dst, remplacement, bilan }
}

const TYPES = { html: 'text/html; charset=utf-8', css: 'text/css; charset=utf-8', js: 'text/javascript; charset=utf-8', json: 'application/json', svg: 'image/svg+xml', avif: 'image/avif', webp: 'image/webp', jpg: 'image/jpeg', png: 'image/png', woff2: 'font/woff2', mp4: 'video/mp4', xml: 'application/xml', txt: 'text/plain' }
async function servir(racine) {
  const srv = http.createServer((q, r) => {
    const p = decodeURIComponent(new URL(q.url, 'http://x').pathname)
    const f = path.join(racine, p.endsWith('/') ? p + 'index.html' : p)
    if (!f.startsWith(racine) || !fs.existsSync(f) || !fs.statSync(f).isFile()) { r.writeHead(404); return r.end() }
    r.writeHead(200, { 'content-type': TYPES[path.extname(f).slice(1)] || 'application/octet-stream', 'cache-control': 'no-store' })
    fs.createReadStream(f).pipe(r)
  })
  await new Promise((ok) => srv.listen(0, '127.0.0.1', ok))
  return { base: `http://127.0.0.1:${srv.address().port}`, fermer: () => new Promise((ok) => srv.close(() => ok())) }
}

// ── injecté avant tout script de la page ─────────────────────────────────────────────────────────────────────────────────────
function INIT(simuler) {
  const natif = CSS.supports.bind(CSS)
  window.__supportsNatif = natif
  if (simuler) {
    const motif = /(animation-timeline|view-timeline|scroll-timeline)/i
    CSS.supports = function (a, b) {
      if (motif.test(String(a))) return false
      return arguments.length > 1 ? natif(a, b) : natif(a)
    }
  }
  const ETATS = new Set(['vu', 'entier', 'solide', 'nav--sombre', 'nav--bande', 'nav--appel', 'nav--retenue', 'tout-rendu', 'js', 'no-js', 'pilote', 'relief-3d', 'menu-ouvert', 'ouvert', 'est-ouverte', 'est-sortie'])
  function sel(el) {
    if (!el || el.nodeType !== 1) return String(el && el.nodeName)
    let s = el.tagName.toLowerCase()
    if (el.id) s += '#' + el.id
    const cls = [...el.classList].filter((c) => !ETATS.has(c))
    if (cls.length) s += '.' + cls.slice(0, 3).join('.')
    const p = el.parentElement
    if (p && !el.id) {
      const freres = [...p.children].filter((x) => x.tagName === el.tagName && x.className === el.className)
      if (freres.length > 1) s += `:${freres.indexOf(el) + 1}`
    }
    return s
  }
  function section(el) {
    if (!el || el.nodeType !== 1) return 'document'
    if (el.closest('header.nav, .panneau, .voile')) return 'nav'
    if (el.closest('.hero--plein')) return 'heros'
    if (el.closest('section.plateau')) return 'plateau'
    const s = el.closest('section[id], footer')
    if (s) return s.id || 'pied'
    if (el === document.documentElement || el === document.body) return 'document'
    return 'hors-section'
  }
  const val = (v) => (v == null ? null : typeof v === 'number' ? Math.round(v * 10) / 10 : String(v))
  window.__anims = function () {
    return document.getAnimations().map((a) => {
      const e = a.effect, t = e && e.target, tl = a.timeline
      const ct = e ? e.getComputedTiming() : {}
      const r = t ? t.getBoundingClientRect() : null
      let sujet = null
      try { if (tl && tl.subject) sujet = sel(tl.subject); else if (tl && tl.source) sujet = sel(tl.source) } catch (x) {}
      return {
        type: a.constructor.name,
        nom: a.animationName || a.transitionProperty || a.id || '',
        cible: sel(t) + (e && e.pseudoElement ? e.pseudoElement : ''),
        section: section(t),
        etat: a.playState,
        timeline: tl ? tl.constructor.name : 'aucune',
        sujet, axe: tl && tl.axis ? tl.axis : null,
        progression: ct.progress == null ? null : Math.round(ct.progress * 1e4) / 1e4,
        tempsCourant: val(a.currentTime),
        duree: val(ct.duration), iterations: ct.iterations === Infinity ? 'infinite' : ct.iterations, delai: val(ct.delay),
        aLEcran: !!(r && r.width > 0 && r.height > 0 && r.bottom > 0 && r.top < innerHeight && r.right > 0 && r.left < innerWidth),
      }
    })
  }
  window.__journal = []
  const mo = new MutationObserver((ms) => {
    for (const m of ms) {
      if (window.__journal.length > 50000) return
      window.__journal.push([sel(m.target), section(m.target), m.attributeName, (m.target.getAttribute(m.attributeName) || '').slice(0, 200)])
    }
  })
  mo.observe(document, { attributes: true, subtree: true, attributeFilter: ['style', 'class', 'hidden', 'open'] })
  window.__lireJournal = function () {
    const agg = new Map()
    for (const [cible, sec, attr, v] of window.__journal) {
      const k = cible + '|' + attr
      const x = agg.get(k) || { cible, section: sec, attribut: attr, n: 0, valeurs: new Set(), derniere: '' }
      x.n++; if (x.valeurs.size < 6) x.valeurs.add(v); x.derniere = v
      agg.set(k, x)
    }
    window.__journal.length = 0
    return [...agg.values()].map((x) => ({ ...x, valeurs: [...x.valeurs] }))
  }
  window.__sel = sel; window.__section = section
}

async function contexte(browser, profil, extra = {}) {
  const base = profil.nom === 'android' ? { ...devices['Pixel 7'] } : { ...devices['iPhone 13'], viewport: { width: 390, height: 844 } }

const rafs = (page, n = 2) => page.evaluate((n) => new Promise((ok) => { let k = 0; const f = () => (++k >= n ? ok() : requestAnimationFrame(f)); requestAnimationFrame(f) }), n)

async function gris(buf) {
  const { data, info } = await sharp(buf).grayscale().raw().toBuffer({ resolveWithObject: true })
  return { data, w: info.width, h: info.height }
}
function ecart(a, b, y0 = 0, y1 = a.h) {
  let s = 0, n8 = 0, n = 0
  for (let y = Math.max(0, y0); y < Math.min(a.h, y1); y++) for (let x = 0; x < a.w; x++) {
    const i = y * a.w + x, d = Math.abs(a.data[i] - b.data[i])
    s += d; if (d > 8) n8++; n++
  }
  return n ? { moyen: Math.round((s / n) * 1000) / 1000, part8: Math.round((n8 / n) * 1e5) / 1e5, pixels: n } : null
}

async function filmer(browser, profil, dossier, depart, nom) {
  const tmp = path.join(TRAVAIL, `videos-${profil.nom}-${nom}`)
  fs.rmSync(tmp, { recursive: true, force: true })
  const taille = profil.nom === 'android' ? { width: 412, height: 839 } : { width: 390, height: 844 }
  const ctx = await contexte(browser, profil, { recordVideo: { dir: tmp, size: taille } })
  const tCreation = Date.now()
  const page = await ctx.newPage()
  await page.goto(profil.url, { waitUntil: 'load', timeout: 60000 })
  await page.evaluate(() => document.fonts.ready)
  if (depart > 0) {
    await page.evaluate((y) => window.scrollTo({ top: y, behavior: 'instant' }), depart)
    await page.waitForTimeout(2500) // tout-rendu, images du point de départ
    await page.evaluate((y) => window.scrollTo({ top: y, behavior: 'instant' }), depart)
  }
  await page.waitForTimeout(2000)
  const tDebut = Date.now()
  const echantillons = await page.evaluate(({ depart, vitesse, duree }) => new Promise((ok) => {
    const t0 = performance.now(), ech = []
    const f = (now) => {
      const t = now - t0
      window.scrollTo({ top: depart + (vitesse * Math.min(t, duree)) / 1000, behavior: 'instant' })
      ech.push([Math.round(t * 10) / 10, Math.round(scrollY * 10) / 10])
      if (t < duree) requestAnimationFrame(f); else ok(ech)
    }
    requestAnimationFrame(f)
  }), { depart, vitesse: VITESSE, duree: DUREE_FILM })
  const tFin = Date.now()
  await page.waitForTimeout(500)
  const video = page.video()
  await ctx.close()
  const brut = await video.path()
  const webm = path.join(dossier, `film-${nom}.webm`)
  fs.copyFileSync(brut, webm)
  const debutS = Math.max(0, (tDebut - tCreation) / 1000 - 0.3)
  const mp4 = path.join(dossier, `film-${nom}.mp4`)
  execFileSync('ffmpeg', ['-y', '-v', 'error', '-ss', debutS.toFixed(2), '-i', webm, '-t', ((tFin - tDebut) / 1000 + 0.6).toFixed(2), '-c:v', 'libx264', '-crf', '20', '-pix_fmt', 'yuv420p', '-movflags', '+faststart', mp4])
  const pas = echantillons.slice(1).map((e, k) => e[1] - echantillons[k][1]).sort((a, b) => a - b)
  const dts = echantillons.slice(1).map((e, k) => e[0] - echantillons[k][0]).sort((a, b) => a - b)
  fs.rmSync(tmp, { recursive: true, force: true })
  return { nom, depart, fichiers: [path.basename(webm), path.basename(mp4)], decoupeDebutS: debutS, imagesRaf: echantillons.length, dureeMesureeMs: echantillons.at(-1)[0], imagesParSeconde: Math.round((echantillons.length / (echantillons.at(-1)[0] / 1000)) * 10) / 10, pasMedianPx: pas[Math.floor(pas.length / 2)], intervalleMedianMs: dts[Math.floor(dts.length / 2)], intervalleP95Ms: dts[Math.floor(dts.length * 0.95)], yDebut: echantillons[0][1], yFin: echantillons.at(-1)[1] }
}

// balayage fin (pas de 40 px, sans capture, page neuve) : où chaque animation liée au défilement avance, et où chaque .reveal / .etape reçoit « vu »
async function balayageFin(browser, profil) {
  const ctx = await contexte(browser, profil)
  const page = await ctx.newPage()
  await page.goto(profil.url, { waitUntil: 'load', timeout: 60000 })
  await page.evaluate(() => document.fonts.ready)
  await page.waitForTimeout(REPOS)
  await page.evaluate(() => { window.__vus = new Set([...document.querySelectorAll('.vu')]) })
  const lignes = []
  for (let y = 0, k = 0; k < 600; k++) {
    await page.evaluate((y) => window.scrollTo({ top: y, behavior: 'instant' }), y)
    await rafs(page, 2)
    const l = await page.evaluate(() => {
      const nouveaux = [...document.querySelectorAll('.vu')].filter((e) => !window.__vus.has(e))
      nouveaux.forEach((e) => window.__vus.add(e))
      const defil = document.getAnimations().filter((a) => a.timeline && a.timeline.constructor.name !== 'DocumentTimeline').map((a) => [a.animationName || a.transitionProperty, window.__sel(a.effect.target) + (a.effect.pseudoElement || ''), window.__section(a.effect.target), a.timeline.constructor.name, Math.round((a.effect.getComputedTiming().progress ?? -1) * 1000) / 1000])
      const liste = document.querySelector('.etapes__liste')
      return { y: Math.round(scrollY), max: document.documentElement.scrollHeight - innerHeight, vus: nouveaux.map((e) => [window.__sel(e), window.__section(e)]), defil, avancee: liste ? liste.style.getPropertyValue('--avancee') : null }
    })
    lignes.push(l)
    if (y >= l.max) break
    y = Math.min(y + 40, l.max)
  }
  await ctx.close()
  // résumé : pour chaque animation liée au défilement, premier y où elle a quitté 0, premier y où elle vaut 1, et les valeurs vues
  const anims = new Map()
  for (const l of lignes) for (const [nom, cible, section, tl, p] of l.defil) {
    const c = nom + ' @ ' + cible + ' @ ' + section
    const x = anims.get(c) || { nom, cible, section, timeline: tl, yDepart: null, yPlein: null, yRetour: null, progressions: 0 }
    if (p > 0 && x.yDepart == null) x.yDepart = l.y
    if (p >= 1 && x.yPlein == null) x.yPlein = l.y
    if (x.yPlein != null && p < 1 && x.yRetour == null) x.yRetour = l.y
    if (p > 0 && p < 1) x.progressions++
    anims.set(c, x)
  }
  const vus = []
  for (const l of lignes) for (const [cible, section] of l.vus) vus.push({ y: l.y, cible, section })
  // la ligne des étapes (kit.js, --avancee en style en ligne, pas une Animation) : premier y hors de 0, premier y à 1
  const av = lignes.filter((l) => l.avancee !== null && l.avancee !== '').map((l) => ({ y: l.y, v: Number(l.avancee) }))
  const avancee = { valeursDistinctes: new Set(av.map((x) => x.v)).size, yPremiereValeur: av.length ? av[0].y : null, yHorsZero: av.find((x) => x.v > 0)?.y ?? null, yPlein: av.find((x) => x.v >= 1)?.y ?? null }
  return { pas: 40, positions: lignes.length, animsDefilement: [...anims.values()], reveals: vus, avancee }
}
