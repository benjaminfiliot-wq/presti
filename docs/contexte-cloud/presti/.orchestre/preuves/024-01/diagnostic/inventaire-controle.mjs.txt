// inventaire-controle.mjs — contrôle croisé de l'inventaire 024-01 (lecture seule du site : rien n'est écrit hors de la sortie JSON).
// Sert les copies générées (natif/, sans-timeline/ et sans-timeline-corrige/ ; les deux dernières avec @supports (animation-timeline: view()|scroll()) réécrits en
// --non-supporte() puis en (animation-timeline-inexistante: view())) et relève, par profil et par page : document.getAnimations() après un défilement complet, les animations calculées
// d'éléments témoins, l'état posé par site.js, les requêtes du relief, de la vidéo, de l'éclaté et de bureau.css.
// Aucune mesure de temps : ce relevé dit seulement ce qui existe et ce qui est absent.
// Usage : node .orchestre/preuves/024-01/diagnostic/inventaire-controle.mjs <racine des copies> <sortie.json>
import http from 'node:http'
import fs from 'node:fs'
import path from 'node:path'
import { chromium, devices } from 'playwright'

const RACINE = path.resolve(process.argv[2])
const SORTIE = path.resolve(process.argv[3])
const MIME = { '.html': 'text/html; charset=utf-8', '.css': 'text/css; charset=utf-8', '.js': 'text/javascript; charset=utf-8', '.json': 'application/json', '.svg': 'image/svg+xml', '.avif': 'image/avif', '.webp': 'image/webp', '.woff2': 'font/woff2', '.mp4': 'video/mp4', '.txt': 'text/plain', '.xml': 'application/xml' }

const serveur = http.createServer((req, res) => {
  const u = decodeURIComponent(new URL(req.url, 'http://x').pathname)
  const f = path.join(RACINE, u)
  if (!f.startsWith(RACINE) || !fs.existsSync(f) || !fs.statSync(f).isFile()) { res.writeHead(404); return res.end() }
  const taille = fs.statSync(f).size, type = MIME[path.extname(f)] || 'application/octet-stream'
  const plage = /bytes=(\d*)-(\d*)/.exec(req.headers.range || '')
  if (plage) {
    const debut = plage[1] ? Number(plage[1]) : 0, fin = plage[2] ? Number(plage[2]) : taille - 1
    res.writeHead(206, { 'Content-Type': type, 'Content-Range': `bytes ${debut}-${fin}/${taille}`, 'Accept-Ranges': 'bytes', 'Content-Length': fin - debut + 1 })
    return fs.createReadStream(f, { start: debut, end: fin }).pipe(res)
  }
  res.writeHead(200, { 'Content-Type': type, 'Content-Length': taille, 'Accept-Ranges': 'bytes' })
  fs.createReadStream(f).pipe(res)
})
await new Promise((ok) => serveur.listen(0, '127.0.0.1', ok))
const BASE = `http://127.0.0.1:${serveur.address().port}`

const SANS_TIMELINE = `(() => {
  const re = /animation-timeline|view-timeline|scroll-timeline/i
  const orig = CSS.supports.bind(CSS)
  const f = function (a, b) { const s = b === undefined ? String(a) : String(a) + ':' + String(b); if (re.test(s)) return false; return b === undefined ? orig(a) : orig(a, b) }
  Object.defineProperty(CSS, 'supports', { value: f, configurable: true, writable: true })
})()`

const iphone = { ...devices['iPhone 13'] }; delete iphone.defaultBrowserType
const pixel = { ...devices['Pixel 7'] }; delete pixel.defaultBrowserType
const PROFILS = [
  { nom: 'chromium-390x844-natif', dossier: 'natif', contexte: { ...iphone, viewport: { width: 390, height: 844 } }, init: null },
  { nom: 'iphone-sans-animation-timeline', dossier: 'sans-timeline', contexte: { ...iphone, viewport: { width: 390, height: 844 } }, init: SANS_TIMELINE },
  { nom: 'android-pixel7', dossier: 'natif', contexte: pixel, init: null },
  // sans-timeline-corrige : les mêmes @supports réécrits en (animation-timeline-inexistante: view()), faux sur Chromium 153 (inventaire-sonde-supports.json)
  { nom: 'iphone-sans-animation-timeline-corrige', dossier: 'sans-timeline-corrige', contexte: { ...iphone, viewport: { width: 390, height: 844 } }, init: SANS_TIMELINE },
  { nom: 'bureau-1440x900-natif', dossier: 'natif', contexte: { viewport: { width: 1440, height: 900 } }, init: null },
]
const PAGES = ['lestraque/index.html', 'lestraque/menuisier-rennes/index.html', 'lestraque/mentions-legales/index.html']

const releve = () => {
  const d = document, h = d.documentElement
  const cle = (a) => {
    const e = a.effect, t = e && e.target
    const nom = a.animationName || (a.transitionProperty ? `transition:${a.transitionProperty}` : a.constructor.name)
    const cible = t ? `${t.tagName.toLowerCase()}${t.id ? '#' + t.id : ''}${t.classList.length ? '.' + [...t.classList].join('.') : ''}${e.pseudoElement || ''}` : '?'
    return { nom, timeline: a.timeline ? a.timeline.constructor.name : 'null', cible, etat: a.playState }
  }
  const groupes = {}
  for (const a of d.getAnimations()) {
    const k = cle(a), g = `${k.nom} | ${k.timeline}`
    groupes[g] = groupes[g] || { n: 0, cibles: [] }
    groupes[g].n++
    if (groupes[g].cibles.length < 3 && !groupes[g].cibles.includes(k.cible)) groupes[g].cibles.push(k.cible)
  }
  const style = (sel, pseudo = null) => {
    const el = d.querySelector(sel)
    if (!el) return null
    const cs = getComputedStyle(el, pseudo)
    return { animationName: cs.animationName, animationTimeline: cs.animationTimeline, display: cs.display, opacity: cs.opacity, content: pseudo ? cs.content : undefined }
  }
  const mm = (q) => matchMedia(q).matches
  const nav = d.querySelector('.nav')
  return {
    url: location.pathname,
    classesHtml: h.className,
    mediaQueries: {
      'telephone.css (max-width:767px) and (min-height:560px) and (prefers-reduced-motion:no-preference)': mm('(max-width:767px) and (min-height:560px) and (prefers-reduced-motion:no-preference)'),
      '(min-width:1024px)': mm('(min-width:1024px)'),
      '(min-width:900px)': mm('(min-width:900px)'),
      '(hover:hover) and (pointer:fine)': mm('(hover:hover) and (pointer:fine)'),
      '(max-width:759px)': mm('(max-width:759px)'),
    },
    cssSupports: { 'animation-timeline: view()': CSS.supports('animation-timeline: view()'), 'animation-timeline: scroll()': CSS.supports('animation-timeline', 'scroll()') },
    saveData: !!(navigator.connection && navigator.connection.saveData),
    fenetre: [innerWidth, innerHeight, devicePixelRatio],
    animations: groupes,
    temoins: {
      'hero img': style('.hero--plein .hero__media img'),
      'hero__media::before': style('.hero--plein .hero__media', '::before'),
      'citation__mot (1er)': style('.citation__mot'),
      'plateau__citation.reveal': style('.plateau__citation.reveal'),
      'atelier__verbe (1er)': style('.atelier__verbe'),
      'eclate__eclate': style('img.eclate__eclate'),
      'eclate__assemble': style('img.eclate__assemble'),
      'eclate__calque (1er)': style('img.eclate__calque'),
      'eclate__pastille (1re)': style('.eclate__pastille'),
      'presta__jauge span': style('.presta__jauge span'),
      'presta__jauge': style('.presta__jauge'),
      'etapes__jauge': style('.etapes__jauge'),
      'etapes__jauge span': style('.etapes__jauge span'),
      'galerie__jauge': style('.galerie__jauge'),
      'galerie__jauge span': style('.galerie__jauge span'),
      'atelier__media img': style('.atelier__media img'),
      'mot (1er)': style('.mot'),
      'compteur__piste': style('.compteur__piste'),
      'etape__num (1er)': style('.etape__num'),
      'ml__in (1er)': style('.ml__in'),
      'presta__ligne.reveal (1re)': style('.presta__ligne.reveal'),
    },
    transitions: {
      'plateau__citation.reveal': (() => { const e = d.querySelector('.plateau__citation.reveal'); return e ? getComputedStyle(e).transitionProperty + ' ' + getComputedStyle(e).transitionDuration : null })(),
      'atelier overflow': (() => { const e = d.querySelector('.atelier'); return e ? getComputedStyle(e).overflow : null })(),
    },
    etatSiteJs: {
      nav: nav ? nav.className : null,
      prestationsSurvol: !!d.querySelector('#prestations.scene--survol'),
      chiffres: (d.querySelector('.chiffres') || {}).className || null,
      travellingPilote: !!d.querySelector('.travelling.pilote'),
      reliefEtat: (d.querySelector('.hero--plein[data-relief]') || { getAttribute: () => null }).getAttribute('data-relief-etat'),
      toileRelief: !!d.querySelector('canvas.hero__relief'),
      videoAtelier: !!d.querySelector('video.atelier__video'),
      premiereQuestionOuverte: !!(d.querySelector('.faq details') || {}).open,
      avanceeEtapes: (d.querySelector('.etapes__liste') || { style: { getPropertyValue: () => '' } }).style.getPropertyValue('--avancee'),
      imgHeroTranslate: (d.querySelector('.hero--plein .hero__media img') || { style: {} }).style.translate || '',
      revealVus: d.querySelectorAll('.reveal.vu').length,
      reveals: d.querySelectorAll('.reveal').length,
      toutRendu: h.classList.contains('tout-rendu'),
    },
  }
}

const resultat = { date: new Date().toISOString(), base: BASE, playwright: (await import('playwright/package.json', { with: { type: 'json' } })).default.version, profils: {} }
const navigateur = await chromium.launch()
try {
  resultat.chromium = navigateur.version()
  for (const p of PROFILS) {
    const ctx = await navigateur.newContext(p.contexte)
    if (p.init) await ctx.addInitScript(p.init)
    resultat.profils[p.nom] = { contexte: { viewport: p.contexte.viewport, dpr: p.contexte.deviceScaleFactor, isMobile: p.contexte.isMobile, hasTouch: p.contexte.hasTouch, userAgent: p.contexte.userAgent, dossier: p.dossier, initSansTimeline: !!p.init }, pages: {} }
    for (const pg of PAGES) {
      const page = await ctx.newPage()
      const requetes = []
      page.on('request', (r) => { const u = r.url(); if (/relief|\/video\/|\/eclate\/|bureau\.css|site\.js/.test(u)) requetes.push(u.replace(BASE, '')) })
      await page.goto(`${BASE}/${p.dossier}/${pg}`, { waitUntil: 'load' })
      await page.waitForTimeout(1500)
      const auChargement = await page.evaluate(releve)
      // défilement complet, par pas de 200 px, pour déclencher IntersectionObserver, tout-rendu et les images paresseuses
      await page.evaluate(async () => {
        const pause = (ms) => new Promise((ok) => setTimeout(ok, ms))
        for (let y = 0; y < document.documentElement.scrollHeight; y += 200) { scrollTo(0, y); await pause(40) }
        scrollTo(0, document.documentElement.scrollHeight); await pause(600)
      })
      const apresDefilement = await page.evaluate(releve)
      resultat.profils[p.nom].pages[pg] = { auChargement, apresDefilement, requetes: [...new Set(requetes)].sort() }
      await page.close()
    }
    await ctx.close()
  }
} finally {
  await navigateur.close()
  serveur.close()
}
fs.mkdirSync(path.dirname(SORTIE), { recursive: true })
fs.writeFileSync(SORTIE, JSON.stringify(resultat, null, 2))
console.log('écrit :', SORTIE)
