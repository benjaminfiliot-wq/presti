async function glisse(id, sel, img, deplacement) {
  const clip = await page.evaluate(({ sel }) => {
    const el = document.querySelector(sel); let t = 0; for (let n = el; n; n = n.offsetParent) t += n.offsetTop
    const H = el.offsetHeight; scrollTo(0, Math.round(t - innerHeight + 0.30 * (innerHeight + H)))
    const r = el.getBoundingClientRect(); return { x: 0, y: Math.max(0, Math.round(r.top)), width: innerWidth, height: Math.round(Math.min(innerHeight, r.bottom) - Math.max(0, r.top)) }
  }, { sel })
  await page.waitForTimeout(400)

  out[id] = { boite: `${clip.width}×${clip.height}`, deplacement, ...comparer(neutre, decale) }
}
