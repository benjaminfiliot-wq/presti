const cache = new Map()
const nomImage = (d, rho, src = 2400) => `d${d.toFixed(5)}_r${rho.toFixed(3)}_${src}.png`
async function imagePng(nom) {
  if (cache.has(nom)) return cache.get(nom)
  const m = nom.match(/^d([\d.]+)_r(-?[\d.]+)_(\d+)\.png$/)
  const png = await versPng(await rendre(+m[1], +m[2], m[3] === '2880' ? SRC2880 : SRC2400))
  cache.set(nom, png); if (cache.size > 8) cache.delete(cache.keys().next().value)
  return png
}
const urlImage = (d, rho, src) => `https://prestiapp.fr/__descente/${nomImage(d, rho, src)}`

const c01 = (x) => Math.min(1, Math.max(0, x))
const lin = (c) => { c /= 255; return c <= 0.04045 ? c / 12.92 : ((c + 0.055) / 1.055) ** 2.4 }
const LIN = Float64Array.from({ length: 256 }, (_, i) => lin(i))
// réduction grise de 160 px et écart : fonctions gris / ecart de cahier/cdc-c2-variantes.mjs (cahier §8.0)
const gris = async (b, z) => { let s = sharp(b); if (z) s = s.extract(z); return (await s.resize(160).greyscale().raw().toBuffer()) }
const ecart = (a, b) => { let s = 0, n8 = 0; for (let i = 0; i < a.length; i++) { const d = Math.abs(a[i] - b[i]); s += d; if (d > 8) n8++ } return { moyen: +(s / a.length / 255).toFixed(4), part8: +(n8 / a.length).toFixed(3) } }
// contraste du fond du titre (cahier §8.0) : titre transparent, PNG, luminance relative des pixels de la boîte de .hero h1, p90, 1,05/(L + 0,05)
async function contraste(png, boite, W, H) {
  const left = Math.max(0, Math.round(boite.x)), top = Math.max(0, Math.round(boite.y))
  const right = Math.min(W, Math.round(boite.x + boite.w)), bottom = Math.min(H, Math.round(boite.y + boite.h))
  if (right - left < 2 || bottom - top < 2) return null
  const { data, info } = await sharp(png).extract({ left, top, width: right - left, height: bottom - top }).removeAlpha().raw().toBuffer({ resolveWithObject: true })
  const L = new Float64Array(data.length / info.channels)
  for (let i = 0, j = 0; i < data.length; i += info.channels, j++) L[j] = 0.2126 * LIN[data[i]] + 0.7152 * LIN[data[i + 1]] + 0.0722 * LIN[data[i + 2]]
  L.sort()
  const p90 = L[Math.floor(0.9 * (L.length - 1))]
  return +(1.05 / (p90 + 0.05)).toFixed(3)
}
// netteté : variance du laplacien, fonction de cahier/cdc-c2-variantes.mjs (section 3) ; zones du cahier (R-5) en 1440×900
const ZONE_BOIS = { left: 900, top: 120, width: 500, height: 300 }, ZONE_MARCHES = { left: 150, top: 120, width: 350, height: 300 }
async function laplacien(png, zone) {
  const { data, info } = await sharp(png).extract(zone).greyscale().raw().toBuffer({ resolveWithObject: true })
  const w = info.width, h = info.height
  let s = 0, s2 = 0, n = 0
  for (let y = 1; y < h - 1; y++) for (let x = 1; x < w - 1; x++) { const i = y * w + x; const v = data[i - 1] + data[i + 1] + data[i - w] + data[i + w] - 4 * data[i]; s += v; s2 += v * v; n++ }
  return +(s2 / n - (s / n) ** 2).toFixed(1)
}
// D-15 : appariement de blocs de 16 px sur la réduction de 160 px, affinage parabolique au sous-pixel
function apparier(A, B, W, H, bx, by, R) {
  const dedans = (u, v) => bx + u >= 0 && by + v >= 0 && bx + u + 16 <= W && by + v + 16 <= H
  const cout = (u, v) => { let s = 0; for (let j = 0; j < 16; j++) for (let i = 0; i < 16; i++) s += Math.abs(A[(by + j) * W + bx + i] - B[(by + v + j) * W + bx + u + i]); return s }
  let best = Infinity, bu = 0, bv = 0
  for (let v = -R; v <= R; v++) for (let u = -R; u <= R; u++) { if (!dedans(u, v)) continue; const c = cout(u, v); if (c < best) { best = c; bu = u; bv = v } }
  const para = (m, c0, p) => { const den = m - 2 * c0 + p; return den > 0 ? 0.5 * (m - p) / den : 0 }
  const du = dedans(bu - 1, bv) && dedans(bu + 1, bv) ? para(cout(bu - 1, bv), best, cout(bu + 1, bv)) : 0
  const dv = dedans(bu, bv - 1) && dedans(bu, bv + 1) ? para(cout(bu, bv - 1), best, cout(bu, bv + 1)) : 0
  return { u: bu + du, v: bv + dv, sadMoyen: +(best / 256).toFixed(2), bordRecherche: Math.abs(bu) === R || Math.abs(bv) === R }
}
const quantile = (arr, p) => { const a = [...arr].sort((x, y) => x - y); return a.length ? a[Math.floor(p * (a.length - 1))] : null }

  for (let k = 1; k < idx.length; k++) { const e = ecart(g[idx[k - 1]], g[idx[k]]).moyen; som += e; max = Math.max(max, e); if (e > 0.085) coupes++; if (e < 0.006) { fige += 1 / 8; figeMax = Math.max(figeMax, fige) } else fige = 0 }
  let max30 = 0; for (let i = 1; i < g.length; i++) max30 = Math.max(max30, ecart(g[i - 1], g[i]).moyen)
  const descente = serie.filter((s) => s.y > 0 && s.y <= 0.3 * H)
  res.mouvementFilm = { images8ips: idx.length, ecartMoyenParImage8ips: r3(som / (idx.length - 1)), ecartMaxPaire8ips: max, pairesAuDessus0085: coupes, plageFigeeLaPlusLongue_s: figeMax, ecartMaxEntreImages30ips: max30, dureePalierDansLeFilm_s: r2(descente.length / IPS), ecartY0contreY270_page: ecart(g[0], g[serie.findIndex((s) => s.y >= 270)]) }
  log('film écrit', sortie)
}

// 3. D-15 et D-9 : y = 0 et y = 0,30 h, animations au repos figées (poussée à 0)
{
  const d270 = deltaDe(film, 270, H), rho270 = rhoDe(film, d270)
  const e0 = await etat(page, { url: film.sans ? null : urlImage(0, 0), y: 0, tPousse: 0 })
  const p0 = await page.screenshot({ type: 'png' })
  const e1 = await etat(page, { url: film.sans ? null : urlImage(d270, rho270), y: 270, tPousse: 0 })
  const p1 = await page.screenshot({ type: 'png' })
  fs.writeFileSync(ICI + `fige-${film.lettre}-y0000.png`, p0); fs.writeFileSync(ICI + `fige-${film.lettre}-y0270.png`, p1)
  res.nettete = { zoneBois: 'x 900-1400, y 120-420', zoneMarches: 'x 150-500, y 120-420', y0: { bois: await laplacien(p0, ZONE_BOIS), marches: await laplacien(p0, ZONE_MARCHES), img: e0.imgSrc }, y270: { bois: await laplacien(p1, ZONE_BOIS), marches: await laplacien(p1, ZONE_MARCHES), img: e1.imgSrc, mediaScale: e1.mediaScale } }
  const A = await gris(p0), B = await gris(p1), WR = 160, HR = 100
  const EXs = 0.404 * 1440, EYs = 0.386 * 1080 - (1080 - 900) * 0.42 // œil à l'écran : 581,76 ; 341,28 (cahier §6)
  const h1Haut = e0.boiteH1.y
  const zones = {
    oeil: { def: 'blocs 16×16 de la réduction entièrement dans le carré de 48 px réduits (432 px écran) centré sur l’œil (581,8 ; 341,3), sous la barre (y ≥ 80) et au-dessus du h1 (y < ' + Math.round(h1Haut) + ')', blocs: [] },
    marches: { def: 'blocs 16×16 entièrement dans x 150-500, y 120-420 (zone des marches du cahier)', blocs: [] },
  }
  const ex = EXs / 9, ey = EYs / 9
  for (let by = 0; by + 16 <= HR; by += 2) for (let bx = 0; bx + 16 <= WR; bx += 2) {
    const dansOeil = bx >= ex - 24 && bx + 16 <= ex + 24 && by >= Math.max(ey - 24, 80 / 9) && by + 16 <= Math.min(ey + 24, h1Haut / 9)
    const dansMarches = bx >= 150 / 9 && bx + 16 <= 500 / 9 && by >= 120 / 9 && by + 16 <= 420 / 9
    if (!dansOeil && !dansMarches) continue
    const m = apparier(A, B, WR, HR, bx, by, 18)
    const cx = (bx + 8) * 9, cy = (by + 8) * 9, vx = cx - EXs, vy = cy - EYs, dist = Math.hypot(vx, vy)
    // prédiction du modèle : rayon écran ρ → 1,04 · r'(ρ/1080, δ) · 1080, tourné de ρ ; zoom du palier autour de l'œil (bloc 2)
    const r = dist / 1080, rp = avance(r, d270) * 1080 * 1.04, th = rho270 * Math.PI / 180
    const px = EXs + (rp / dist) * (vx * Math.cos(th) - vy * Math.sin(th)), py = EYs + (rp / dist) * (vx * Math.sin(th) + vy * Math.cos(th))
    const b = { centre: [cx, cy], distanceOeil: r1(dist), mesure_px: [r1(m.u * 9), r1(m.v * 9)], module_px: r1(Math.hypot(m.u, m.v) * 9), radial_px: r1((m.u * vx + m.v * vy) / dist * 9), modele_px: [r1(px - cx), r1(py - cy)], moduleModele_px: r1(Math.hypot(px - cx, py - cy)), sadMoyen: m.sadMoyen, bordRecherche: m.bordRecherche }
    if (dansOeil) zones.oeil.blocs.push(b)
    if (dansMarches) zones.marches.blocs.push(b)
  }
  for (const z of Object.values(zones)) {
    const mod = z.blocs.map((b) => b.module_px), rad = z.blocs.map((b) => b.radial_px), modM = z.blocs.map((b) => b.moduleModele_px)
    z.resume = { blocs: z.blocs.length, moduleMedian_px: r1(quantile(mod, 0.5)), moduleP90_px: r1(quantile(mod, 0.9)), radialMedian_px: r1(quantile(rad, 0.5)), moduleModeleMedian_px: r1(quantile(modM, 0.5)), blocsAuBordDeLaRecherche: z.blocs.filter((b) => b.bordRecherche).length }
  }
  res.perceptibiliteD15 = { methode: 'réduction grise 160×100 (gris de cdc-c2-variantes.mjs), y = 0 contre y = 270, poussée figée à 0 ; SAD, recherche ±18 px réduits (±162 px écran), pas des blocs 2 px réduits, affinage parabolique ; px écran = px réduits × 9', delta: d270, rho_deg: rho270, zones }
  log('D-15', JSON.stringify({ oeil: zones.oeil.resume, marches: zones.marches.resume }))
}
