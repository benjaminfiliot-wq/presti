const ffprobe = (f) => {
  const j = JSON.parse(execFileSync('ffprobe', ['-v', 'error', '-count_frames', '-show_entries', 'stream=codec_type,codec_name,width,height,pix_fmt,r_frame_rate,nb_read_frames:format=duration,size', '-of', 'json', ICI + f], { encoding: 'utf8' }))
  const v = j.streams.find((s) => s.codec_type === 'video')
  return { fichier: f, duree_s: +(+j.format.duration).toFixed(3), octets: +j.format.size, codec: v.codec_name, format: `${v.width}x${v.height}`, pixfmt: v.pix_fmt, ips: v.r_frame_rate, images: +v.nb_read_frames, pistesAudio: j.streams.filter((s) => s.codec_type === 'audio').length }
}
