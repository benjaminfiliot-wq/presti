import { createRequire } from 'node:module'
import { execFileSync } from 'node:child_process'
const require = createRequire('C:/Users/benja/Documents/presti/package.json')
const sharp = require('sharp')
const ICI = 'C:/Users/benja/AppData/Local/Temp/claude/C--Users-benja-Documents-ofm/fdff3dee-5b68-4d0d-a1d7-d3b56ef3740d/scratchpad/signature/films/descente/'
const v = execFileSync('ffmpeg', ['-v', 'error', '-i', ICI + 'descente-sansgpu.mp4', '-ss', (81.5 / 30).toFixed(4), '-frames:v', '1', '-f', 'rawvideo', '-pix_fmt', 'gray', '-'], { maxBuffer: 1 << 28 })

const W = 1440, H = 900, d = Buffer.alloc(W * H)

for (const r of cell) console.log([...r].map((x) => x.toFixed(0).padStart(3)).join(''))
// signe : le film est-il un zoom plus fort / plus faible ? profil horizontal moyen sur la bande y 120-420
await sharp(d, { raw: { width: W, height: H, channels: 1 } }).jpeg({ quality: 85 }).toFile(ICI + 'diag-difference-y270.jpg')
