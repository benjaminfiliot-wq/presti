def clean(s):
    return html.escape(str(s), quote=True)

def slug(s):
    s = unicodedata.normalize('NFD', s).encode('ascii', 'ignore').decode().lower()
    return re.sub(r'[^a-z0-9]+', '-', s).strip('-')

def page(row):
    métier = row['metier']; s = spec[métier]
    model = json.loads((MODELS / s['model'] / 'contenu.json').read_text())
    name, town, phone = row['nom'], row['commune'], row['telephone'].strip()
    title = f"{s['label']} à {town} — {name}"
    # Les modèles Presti fournissent l'ossature, le chapeau et la prudence du texte.
    assert model['pages']['/']['sections'][0]['genre'] == 'prestations'
    qualifier = 'Les prestations propres à cette entreprise seront précisées ensemble.'
    phone_link = f'<a href="tel:{re.sub(r"[^0-9+]", "", phone)}">{clean(phone)}</a>' if phone else '<span>Coordonnées à compléter</span>'
    phone_label = 'Téléphone professionnel' if phone else 'Contact'
    contact_button = f'<a class="button" href="tel:{re.sub(r"[^0-9+]", "", phone)}">Appeler l’entreprise</a>' if phone else '<a class="button" href="#contact">Voir la présentation</a>'
    services = ''.join(f'<div class="service"><span class="dot"></span><span>{clean(x)}</span><span class="arrow">↗</span></div>' for x in s['items'])
    style = CSS.replace('ACCENT',s['accent']).replace('PALE',s['pale'])
    # Conserver le nom brut dans le texte pour que la page soit cherchable, y compris les esperluettes.
    name_text = name.replace('<','&lt;').replace('>','&gt;')
    return f'''<!doctype html>
<html lang="fr"><head><meta charset="utf-8"><meta name="viewport" content="width=device-width, initial-scale=1"><meta name="color-scheme" content="light"><title>{clean(title)}</title><meta name="description" content="Proposition de page d’accueil pour {clean(name)}, {clean(s['label'].lower())} à {clean(town)}."><style>{style}</style></head>
<body><header class="mast"><div class="wrap"><a class="brand" href="#haut">{name_text}</a><nav class="nav" aria-label="Navigation"><a href="#prestations">Présentation</a><a class="contact-link" href="#contact">Contact ↗</a></nav></div></header>
<main id="haut"><section class="hero"><div class="wrap hero-grid"><div class="hero-copy"><div class="eyebrow">{clean(s['article'])} · {clean(town)}</div><h1>{clean(s['label'])}<span>à {clean(town)}</span></h1><p class="lead">{clean(s['line'])}. {clean(s['lead'])}</p><div class="hero-actions">{contact_button}<a class="text-link" href="#prestations">Découvrir la page ↗</a></div><div class="location"><b>{name_text}</b> · {clean(town)}</div></div><div class="visual">{art(s['motif'])}<span class="visual-caption">Esquisse · {clean(s['label'])}</span></div></div></section>

Les textes et les prestations sont à valider avec l’entreprise.</p></div><div class="contact-box"><small>{phone_label}</small>{phone_link}</div></div></section></main>
<footer class="footer"><div class="wrap"><span>{name_text} · {clean(town)}</span><span class="disclaimer">Maquette — exemple non contractuel</span></div></footer></body></html>'''

def main():
    rows = list(csv.DictReader((ROOT / 'prospection/liste-2026-09-27.csv').open(), delimiter=';'))
    eligible = sorted((r for r in rows if r['a_un_site'] in ('non','inconnu')), key=lambda r: -float(r['score_laya']))
    chosen, counts = [], Counter()
    # Minimum deux par métier, puis meilleurs scores avec quatre places au maximum.
    for row in eligible:
        if counts[row['metier']] < 2:
            chosen.append(row); counts[row['metier']] += 1
    for row in eligible:
        if row not in chosen and counts[row['metier']] < 4 and len(chosen) < 30:
            chosen.append(row); counts[row['metier']] += 1
    chosen.sort(key=lambda r: -float(r['score_laya']))
    assert len(chosen) == 30
    manifest = []
    for n, row in enumerate(chosen, 1):
        folder = f"{n:02d}-{slug(row['nom'])}"
        dest = OUT / folder
        dest.mkdir(exist_ok=True)
        (dest / 'index.html').write_text(page(row), encoding='utf-8')
        manifest.append(dict(n=n, dossier=folder, siren=row['siren'], entreprise=row['nom'], metier=row['metier'], commune=row['commune'], modele=spec[row['metier']]['model'], score_laya=float(row['score_laya'])))
    (OUT / 'manifeste.json').write_text(json.dumps(manifest, ensure_ascii=False, indent=2) + '\n', encoding='utf-8')
    lines = ['# Maquettes du 27 septembre 2026', '', '## Ordre d’appel conseillé', '', 'Classement par `score_laya` décroissant parmi les entreprises sans site connu (`non` ou `inconnu`). Pour varier les métiers : au moins deux et au plus quatre entreprises de chaque métier. L’ordre ci-dessous est celui du manifeste ; aucun appel ni envoi n’a été effectué.', '', '| Rang | Entreprise | Métier | Commune | Téléphone publié | Score |', '| ---: | --- | --- | --- | --- | ---: |']
    for n,r in enumerate(chosen,1):
        lines.append(f"| {n} | {r['nom']} | {r['metier']} | {r['commune']} | {r['telephone'].strip() or 'Non renseigné dans la liste'} | {r['score_laya']} |")
    lines += ['', '## Modèle utilisé par métier', '', '| Métier | Modèle Presti | Adaptation |', '| --- | --- | --- |']
    for métier,s in spec.items():
        adapt = 'Vocabulaire du modèle conservé et ajusté à la présentation.' if métier in ('menuisier','électricien','plombier','couvreur') else f"Titre, texte d’introduction, rubriques et illustration réécrits pour le métier de {métier}."
        lines.append(f"| {métier} | `{s['model']}` | {adapt} |")
    lines += ['', '## Adaptations et limites', '', 'Les pages reprennent la structure éditoriale des modèles de Presti : métier et commune dans le titre, chapeau de présentation, pistes de prestations et prise de contact. Les illustrations vectorielles sont décoratives et intégrées au HTML. Aucun contenu distant ni photo de personne n’est chargé. La mention « Maquette — exemple non contractuel » figure en bas de chaque page.', '', 'Les rubriques de prestations sont des exemples à valider après échange avec l’entreprise. Elles ne prouvent pas que chaque entreprise réalise ces travaux. Les noms, communes, métiers, scores et éventuels téléphones proviennent exclusivement de la liste CSV. Les coordonnées absentes ne sont pas inventées. Aucune certification RGE, expérience chiffrée ou preuve sociale n’est affirmée.', '', 'Les maquettes sont des fichiers locaux ; elles ne sont ni publiées ni envoyées. Les captures montrent un état visuel, pas un site final. L’entreprise devra valider ses services, ses textes, ses coordonnées et son accord avant tout usage public.', '', '## Vérification visuelle', '', 'Cinq captures ont été ouvertes et examinées visuellement : bureau des rangs 1 (paysagiste) et 8 (carreleur), téléphone des rangs 4 (paysagiste, nom long), 22 (isolation) et 30 (chauffagiste, commune longue). Les noms, le métier, la commune, les titres et les boutons sont lisibles ; les illustrations restent dans leur cadre. Sur téléphone, les communes longues passent sur plusieurs lignes sans débordement horizontal apparent. Les captures montrent le haut des pages ; les sections inférieures ont été contrôlées dans le HTML et restent à valider avec chaque entreprise avant usage.']
    (OUT / 'LISEZMOI.md').write_text('\n'.join(lines) + '\n', encoding='utf-8')

if __name__ == '__main__': main()
