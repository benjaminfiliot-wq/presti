/* recit.bureau.css — famille « recit » (vague 5, 021-03) : le récit du plateau et de l'atelier, au bureau chorégraphié seulement.
   Liée par generer.mjs avec media="(min-width: 1024px)" ; tout vit en plus sous prefers-reduced-motion:no-preference et
   @supports (animation-timeline: view()), comme le bloc v26 de kit.css qu'elle prolonge. Toutes les règles portent html.js :
   sans JavaScript, cette feuille est quand même chargée au bureau, et rien ne doit y changer (ni mots pâles, ni chiffres cachés).
   Hors de ce bloc, les spans de recit.mjs n'ont aucun style. Propriétés animées : opacity et transform seulement. */
@media (min-width:1024px) and (prefers-reduced-motion:no-preference){
  @supports (animation-timeline: view()){
    /* A. La citation s'allume mot à mot au rythme du défilement, de 0,55 (le plancher des numéros d'étapes : 5,49:1 mesuré au
       texte translucide) à 1, sur la timeline du paragraphe. Plage et pas de 3 % des titres qui se lèvent (mot-leve) : le 15e
       mot est plein à cover 66 %. La figure ne monte plus : le bloc qui monte et les mots qui s'allument se brouillaient. */
    html.js .plateau__citation p{view-timeline:--citation block}
    html.js .citation__mot{animation:citation-allume linear both;animation-timeline:--citation;animation-range:cover calc(2% + var(--i,0) * 3%) cover calc(24% + var(--i,0) * 3%)}
    html.js .plateau__citation.reveal:has(.citation__mot){animation:none;opacity:1;transform:none}

    /* C. Les quatre verbes de l'atelier s'allument et remontent de .12 em, au pas de 6 % de la cascade des .reveal : le 4e est
       plein à cover 52 %, avant le milieu de l'écran. inline-block : une transformation ne s'applique pas à un span en ligne. */
    html.js .atelier__phrase{view-timeline:--phrase block}
    html.js .atelier__verbe{display:inline-block;animation:verbe-allume linear both;animation-timeline:--phrase;animation-range:cover calc(20% + var(--i,0) * 6%) cover calc(34% + var(--i,0) * 6%)}

    /* B. Les chiffres attendent que la citation soit lue (recit.js pose chiffres--attente) : opacité 0 seulement — ni display, ni
       visibility, ni aria-hidden, le bloc reste lu par les lecteurs d'écran. Puis le bloc entre (0,55 s) et les nombres se comptent
       au temps, jamais au défilement. */
    html.js .chiffres.chiffres--attente{opacity:0}
    html.js .chiffres.chiffres--entre{animation:chiffres-entre .55s cubic-bezier(.16,1,.3,1) both}
    /* le dévoilement du kit (monte, lié au défilement) jouait encore sur chaque chiffre quand le bloc entre : le 3e chiffre était à
       70 % d'opacité et décalé pendant son comptage si l'on s'arrêtait là. Un seul mouvement : celui du bloc. */
    html.js .chiffres:has(.chiffre__compte) .chiffre.reveal{animation:none;opacity:1;transform:none}
    /* Largeur immobile pendant le comptage (de 0 à 180, on passe d'un chiffre à trois) : la vraie valeur garde sa place dans la
       mise en page, juste transparente ; le compte se pose par-dessus, aligné à droite, donc collé à son unité. */
    html.js .chiffre__valeur{position:relative}
    html.js .chiffre__compte{position:absolute;top:0;right:0;white-space:nowrap;font-variant-numeric:lining-nums tabular-nums}
    html.js .chiffres--compte .chiffre__vrai{opacity:0}

    @keyframes citation-allume{from{opacity:.55}to{opacity:1}}
    @keyframes verbe-allume{from{opacity:.55;transform:translateY(.12em)}to{opacity:1;transform:none}}
    @keyframes chiffres-entre{from{opacity:0;transform:translateY(1rem)}to{opacity:1;transform:none}}
  }
}
