/* telephone.css — famille « telephone » (vague 5, 021-17 ; Q4 tranchée « oui, sans les cartes des prestations » le 15/09) : le
   profil léger du téléphone portrait, en CSS seul, ajouté à styles.css après kit.css et les autres familles. Aujourd'hui, deux
   animations seulement tournent au téléphone et le héros paraît figé ; ici, les trois gestes du bureau reviennent sous une forme
   qui ne coûte rien : la cage tourne dans le héros, la citation et les verbes s'allument au défilement, l'éclaté des étapes s'ouvre
   en entrant. Rien que transform (rotate, scale, translate) et opacity, qui passent par le compositeur : ni HTML ni JavaScript
   nouveau, ni worker, ni WebGL, aucune requête de plus au chargement.
   Tout vit sous téléphone portrait, mouvement permis et html.js : la tablette (768 à 1 023 px, photo paysage et pousse), le
   mouvement réduit, la page sans JavaScript, le bureau et les pages communes ne changent pas. Les gestes liés au défilement vivent
   en plus sous @supports (animation-timeline: view()) : sans view(), le texte se lit plein et l'éclaté reste l'image fixe
   d'eclate.css. */
@media (max-width:767px) and (min-height:560px) and (prefers-reduced-motion:no-preference){
  /* 1. RL au héros, à la place de la dérive (kit.css, derive-tel-haut) : la cage tourne et le projecteur balaie le bois, avec les
     horloges du bureau (rotation 10 s aller-retour sinusoïdal, projecteur 9 s linéaire décalé de −2,5 s, un quart de période).
     L'œil de la spirale reste affiché où la dérive le laissait : 61 % + 15,86 % = 76,86 % de la hauteur, comme 61 % × 1,26 depuis
     le haut, quelle que soit l'échelle ; la rotation pivote sur lui, donc il ne bouge pas.
     ±2,25° (signature-tourne-2, l'amplitude du bureau en 34/25) et non ±4° : à 4°, le premier passage du juge a trouvé des pixels
     isolés découverts au cœur de la photo (3 px à 7,5 s, TL-2), là où la géométrie couvre avec plus de 30 px de marge. Ce sont des
     fissures du compositeur sur la couche tournée, présentes aussi en capture à la densité de l'écran, qu'aucun réglage à 4° n'a
     fait disparaître (will-change, backface-visibility, rotation du picture, échelles 1,32 et 1,33). À 2,25° : 0 pixel sur les
     201 instants du juge. 1,29 couvre les coins à ±2,25° sur tout le portrait (marge la plus courte : 6,4 px en 390×844, 4 px en
     375×667, 3,3 px en 767×1024). Le raccourci animation remplace derive-tel-haut et pousse : rien d'autre ne tourne au héros.
     signature-tourne-2 et signature-projecteur sont déclarés hors de tout @media dans kit.css. */
  html.js.da--signature .ouverture>.hero--plein:not(.hero--commune) .hero__media img{transform-origin:50% 61%;translate:0 15.86%;scale:1.29;animation:signature-tourne-2 10s cubic-bezier(.37,0,.63,1) infinite alternate}
  /* Le projecteur du bureau, recopié tel quel (kit.css, v28 L1). Seule sa géométrie change, centrée sur l'œil affiché : rayon de
     180 vw (702 px en 390 px de large), le coin le plus loin de l'œil en est à 678 px, 97 % du rayon ; le masque éteint la lumière
     vers le bord, donc en haut, sous le titre. .hero coupe ce qui déborde (overflow:hidden) : aucun défilement horizontal. */
  html.js.da--signature .ouverture>.hero--plein:not(.hero--commune) .hero__media::before{content:"";position:absolute;z-index:1;pointer-events:none;left:calc(50% - 180vw);top:calc(76.86% - 180vw);width:360vw;height:360vw;background:conic-gradient(from 0deg,rgba(255,214,150,0.4) 0deg,rgba(255,214,150,0.2) 25deg,rgba(255,214,150,0) 45deg,rgba(10,7,4,0) 45deg,rgba(10,7,4,0.475) 65deg,rgba(10,7,4,0.95) 90deg,rgba(10,7,4,0.475) 115deg,rgba(10,7,4,0) 135deg,rgba(255,214,150,0) 135deg,rgba(255,214,150,0.2) 155deg,rgba(255,214,150,0.4) 180deg,rgba(255,214,150,0.2) 205deg,rgba(255,214,150,0) 225deg,rgba(10,7,4,0) 225deg,rgba(10,7,4,0.475) 245deg,rgba(10,7,4,0.95) 270deg,rgba(10,7,4,0.475) 295deg,rgba(10,7,4,0) 315deg,rgba(255,214,150,0) 315deg,rgba(255,214,150,0.2) 335deg,rgba(255,214,150,0.4) 360deg);-webkit-mask:radial-gradient(closest-side,transparent 5%,#000 17%,#000 72%,transparent 100%);mask:radial-gradient(closest-side,transparent 5%,#000 17%,#000 72%,transparent 100%);mix-blend-mode:soft-light;animation:signature-projecteur 9s linear -2.5s infinite}
  /* le voile du téléphone (kit.css, v20) reste au-dessus du projecteur : le titre garde son fond sombre */
  html.js.da--signature .ouverture>.hero--plein:not(.hero--commune) .hero__media::after{z-index:2}

  @supports (animation-timeline: view()){
    /* 2. La citation s'allume mot à mot, de 0,55 à 1, sur la timeline de son paragraphe : la plage et le pas de 3 % du bureau
       (recit.bureau.css), le 15e mot plein à cover 66 %. La figure ne monte plus pendant que les mots s'allument (un seul mouvement
       à la fois) : le .reveal du kit est neutralisé, transition comprise, sans quoi kit.js la ferait encore glisser. */
    html.js .plateau__citation blockquote p:has(.citation__mot){view-timeline:--telephone-citation block}
    html.js .plateau__citation .citation__mot{animation:telephone-mot linear both;animation-timeline:--telephone-citation;animation-range:cover calc(2% + var(--i,0) * 3%) cover calc(24% + var(--i,0) * 3%)}
    html.js .plateau__citation.reveal:has(.citation__mot){opacity:1;transform:none;transition:none}

    /* 3. Les quatre verbes de l'atelier s'allument et remontent de .12 em, au pas de 6 % : le 4e plein à cover 52 %.
       .atelier est en overflow:hidden (kit.css), ce qui en fait un conteneur de défilement : la timeline de la phrase s'y
       rattacherait et resterait figée. clip coupe pareil sans défiler (la même correction qu'au bureau, kit.css v26).
       inline-block : translate ne s'applique pas à un span en ligne. */
    html.js .atelier{overflow:clip}
    html.js .atelier__phrase:has(.atelier__verbe){view-timeline:--telephone-verbes block}
    html.js .atelier__verbe{display:inline-block;animation:telephone-verbe linear both;animation-timeline:--telephone-verbes;animation-range:cover calc(20% + var(--i,0) * 6%) cover calc(34% + var(--i,0) * 6%)}
    /* Un verbe pâle reste lisible : au téléphone, les verbes (21,6 px, graisse 300) demandent 4,5:1 à 0,55 d'opacité, et le voile
       du kit (kit.css, .6 → .3 à 30 %) ne donnait que 3,22 à 3,50 sur la photo, à 29 % de la hauteur de la section où tombe leur
       ligne (juge, RC-9). Le haut du voile tient plus longtemps (.66 → .56 à 33 %, éteint à 54 %) : 5,35 à 5,61, mesuré par la
       méthode du juge ; la main et le papier de verre, plus bas, gardent leur lumière. Seulement quand les verbes s'allument. */
    html.js .atelier:has(.atelier__verbe) .atelier__media::after{background:linear-gradient(180deg,rgba(10,7,4,.66) 0%,rgba(10,7,4,.56) 33%,rgba(10,7,4,0) 54%),linear-gradient(0deg,rgba(10,7,4,.48) 0%,rgba(10,7,4,0) 18%)}

    /* 4. L'éclaté s'ouvre en entrant à l'écran et se referme en remontant (eclate.mjs pose les calques et leur vecteur --vx, --vy,
       en % de la scène). Timeline nommée sur la figure, jamais view() sur les calques : .etapes__photo est en overflow:hidden
       (kit.css), les calques s'y rattacheraient en restant figés. La boîte du rendu devient conteneur de taille (4:3 gardé) pour
       que les calques se déplacent en cqw et cqh. Au téléphone, assemble.avif et les calques remplacent eclate.avif, qui n'est
       plus demandée : les images paresseuses en display:none ne partent jamais. */
    /* kit.js rend toutes les sections lointaines dès le premier défilement avec html.tout-rendu. Sans une seconde borne sur la
       figure, Chromium découvre donc ses images lazy 3 656 px sous l'écran, 600 px avant la limite mesurée par le témoin natif
       du juge (3 056 px). La figure garde sa propre visibilité différée et sa boîte 4:3 : le chargement reste paresseux sans
       changer sa géométrie ni sa timeline quand elle approche réellement de l'écran. */
    html.js #etapes figure.eclate{content-visibility:auto;contain-intrinsic-size:auto 292.5px;view-timeline:--telephone-eclate block}
    html.js #etapes .eclate__rendu{container-type:size}
    html.js #etapes figure.eclate img.eclate__eclate{display:none}
    html.js #etapes figure.eclate img.eclate__assemble{display:block;left:0;top:0;width:100%;height:100%;animation:telephone-efface linear both;animation-timeline:--telephone-eclate;animation-range:cover 11% cover 12%}
    /* Aucune image en double, aucun creux : les calques deviennent opaques sous l'image assemblée encore pleine (cover 10 → 11 %),
       elle s'efface ensuite sur des calques identiques (11 → 12 %), puis seulement ils partent (12 → 45 %). À cover 10 %, ils sont
       donc immobiles à leur place. Linéaire : c'est le défilement qui donne le rythme. Les pastilles arrivent l'éclaté ouvert. */
    html.js #etapes figure.eclate img.eclate__calque{display:block;animation:telephone-apparait linear both,telephone-eclate linear both;animation-timeline:--telephone-eclate,--telephone-eclate;animation-range:cover 10% cover 11%,cover 12% cover 45%}
    html.js #etapes figure.eclate .eclate__pastille{animation:telephone-apparait linear both;animation-timeline:--telephone-eclate;animation-range:cover 45% cover 49%}
  }
}
