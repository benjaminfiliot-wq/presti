// Adaptation locale de la copie de generer.mjs : sites privés d'entreprises réelles.
import fs from 'node:fs'
import path from 'node:path'
import { fileURLToPath } from 'node:url'
const here=path.dirname(fileURLToPath(import.meta.url)), root=path.dirname(here)
const da=JSON.parse(fs.readFileSync(path.join(here,'_kit/da.json')))
const esc=s=>String(s??'').replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;').replace(/"/g,'&quot;')
const slug=s=>String(s).normalize('NFD').replace(/[\u0300-\u036f]/g,'').toLowerCase().replace(/[^a-z0-9]+/g,'-').replace(/^-|-$/g,'')
const tel=s=>'tel:+33'+String(s).replace(/\D/g,'').slice(1)
const labels={plombier:'Plomberie',menuisier:'Menuiserie',carreleur:'Carrelage',macon:'Maçonnerie',couvreur:'Couverture',peintre:'Peinture',electricien:'Électricité'}
const sections={plombier:['Installation','Rénovation','Dépannage'],menuisier:['Agencement','Menuiseries','Ouvrages sur mesure'],carreleur:['Sols','Murs','Pièces d’eau'],macon:['Maçonnerie générale','Rénovation','Aménagements'],couvreur:['Couverture','Zinguerie','Entretien de toiture'],peintre:['Peinture','Préparation des supports','Revêtements'],electricien:['Installation','Rénovation','Équipements']}
const licenceUrl=s=>s==='CC0'?'https://creativecommons.org/publicdomain/zero/1.0/deed.fr':`https://creativecommons.org/licenses/${s.toLowerCase().replace(/^cc\s+/,'').replace(/\s+(\d+(?:\.\d+)?)$/,'/$1')}/deed.fr`
function render(dir){
 const src=path.join(root,dir),out=path.join(src,'site')
 const site=JSON.parse(fs.readFileSync(path.join(src,'site.json'))), content=JSON.parse(fs.readFileSync(path.join(src,'contenu.json'))), attrs=JSON.parse(fs.readFileSync(path.join(src,'attributions.json')))
 const art=da[site.metier], name=site.nom_affiche, city=site.commune, label=labels[site.metier], phone=site.telephone
 if(!art||!name||!content.pages||!site.pages)throw Error(`${dir}: données incomplètes`)
 const files=attrs.map(a=>path.basename(a.fichier));for(const f of files)if(!fs.existsSync(path.join(out,'images',f)))throw Error(`${dir}: image absente ${f}`)
 const hero=attrs.find(a=>path.basename(a.fichier)==='hero-1600.webp')||attrs[0], rest=attrs.filter(a=>a!==hero)
 const image=(a,prefix='',cl='')=>`<img class="${cl}" src="${prefix}${esc(a.fichier)}" alt="Photo d’illustration : ${esc(a.alt||a.legende)}" loading="${a===hero?'eager':'lazy'}">`
 const credit=a=>`${esc(a.auteur)} · ${esc(a.licence)} · illustration`
 const linked=(p,base='')=>base+p.chemin.slice(1)
 const pageFor=c=>site.pages.find(p=>p.type==='commune'&&p.cible===c)
 const communeLinks=(base='')=>site.communes.slice(1).map(c=>`<a href="${linked(pageFor(c),base)}">${esc(c)} <span aria-hidden="true">↗</span></a>`).join('')
 const home=content.pages['/'], intro=home.sections.find(s=>s.genre==='prestations')?.blocs?.find(b=>b.type==='p')?.texte||home.chapeau
 const serviceBlocs=content.pages['/prestations/'].sections[0].blocs;const services=[];for(let i=0;i<serviceBlocs.length;i+=2)services.push({name:serviceBlocs[i].texte,detail:serviceBlocs[i+1].texte})
 const plate=`<div class="ref-note">Maquette — exemple non contractuel · Projet privé non publié</div>`
 function shell(page,body){
  const base=page.chemin==='/'?'':'../', title=content.pages[page.chemin]?.title||`${name} | Maquette`,desc=content.pages[page.chemin]?.description||`Maquette privée pour ${name}`
  return `<!doctype html><html lang="fr"><head><meta charset="utf-8"><meta name="viewport" content="width=device-width,initial-scale=1"><meta name="robots" content="noindex,nofollow,noarchive"><meta name="denomination-sirene" content="${esc(site.nom)}"><title>${esc(title)}</title><meta name="description" content="${esc(desc)}"><link rel="preload" href="${base}fonts/${slug(art.display)}.woff2" as="font" type="font/woff2" crossorigin><link rel="stylesheet" href="${base}kit.css"><link rel="stylesheet" href="${base}styles.css"></head><body>${plate}<header class="ref-nav"><div class="ref-wrap ref-nav__in"><a class="ref-brand" href="${base}index.html">${esc(name)}</a><nav aria-label="Navigation principale"><a href="${base}index.html">Accueil</a><a href="${base}prestations/">Prestations</a><a href="${base}contact/">Contact</a></nav><a class="ref-nav__call" href="${tel(phone)}">${esc(phone)}</a></div></header><main>${body}</main><footer class="ref-footer"><div class="ref-wrap ref-footer__grid"><div><strong>${esc(name)}</strong><p>${esc(label)} · ${esc(city)}</p><p>Site de démonstration préparé pour un échange avec l’entreprise. La zone d’intervention et les prestations restent à confirmer.</p></div><div><strong>Explorer</strong><a href="${base}index.html">Accueil</a><a href="${base}prestations/">Prestations</a><a href="${base}contact/">Contact</a><a href="${base}mentions/">Mentions de la maquette</a></div><div><strong>Communes proches</strong>${communeLinks(base)}</div></div><div class="ref-wrap ref-footer__base">Maquette — exemple non contractuel · Photos de démonstration créditées dans les mentions · Aucun message n’est collecté ni envoyé.</div></footer><script src="${base}kit.js" defer></script></body></html>`
 }
 const cardServices=services.map((s,i)=>`<article class="ref-service reveal"><span class="ref-num">0${i+1}</span><h3>${esc(s.name)}</h3><p>${esc(s.detail)}</p></article>`).join('')
 const gallery=rest.slice(0,3).map(a=>`<figure class="reveal">${image(a)}<figcaption>${credit(a)}</figcaption></figure>`).join('')
 const contact=`<section class="ref-contact" id="contact"><div class="ref-wrap"><span class="ref-kicker">Contact direct</span><h2>Parlons de votre projet</h2><p>Pour connaître les prestations proposées et la zone d’intervention, appelez directement l’entreprise. Le numéro provient d’une source publiée et reste à confirmer au premier appel.</p><a class="ref-phone" href="${tel(phone)}">${esc(phone)}</a><a class="ref-link" href="contact/">Voir la page contact <span aria-hidden="true">↗</span></a></div></section>`
 const homeBody=`<section class="ref-hero ref-hero--${art.hero}">${image(hero,'','ref-hero__img')}<div class="ref-wrap ref-hero__in"><span class="ref-kicker">${esc(label)} · ${esc(city)}</span><h1>${esc(home.h1)}<em>${esc(home.h1Ligne||'')}</em></h1><p>${esc(home.chapeau)} Une proposition pour ${esc(name)}, à préciser avec l’entreprise.</p><a class="ref-button" href="prestations/">Découvrir les prestations <span aria-hidden="true">↗</span></a></div><span class="ref-hero__credit">${credit(hero)}</span></section><section class="ref-intro"><div class="ref-wrap ref-two"><div><span class="ref-kicker">La proposition</span><h2>Un métier, une présence locale</h2></div><div><p>${esc(intro)}</p><p>La maquette donne une place claire au métier, aux images et au contact. Les prestations exactes, les secteurs d’intervention et les photos propres à l’entreprise seront confirmés avec elle.</p><p>Ce premier écran indique le métier et la commune. La suite propose un chemin lisible : comprendre les rubriques possibles, voir une ambiance visuelle, puis trouver le contact.</p></div></div></section><section class="ref-section" id="prestations"><div class="ref-wrap"><span class="ref-kicker">Explorer le métier</span><h2>Les prestations possibles</h2><p>Ces rubriques illustrent une structure de site. Leur contenu ne décrit pas encore l’offre confirmée de ${esc(name)}.</p><div class="ref-services">${cardServices}</div><a class="ref-textlink" href="prestations/">Voir la page prestations ↗</a></div></section><section class="ref-section ref-section--dark"><div class="ref-wrap"><span class="ref-kicker">Ambiance visuelle</span><h2>Images du métier</h2><p>Photos de démonstration sous licence, sans lien avec les chantiers de ${esc(name)}. Les photos de l’entreprise seront choisies avec son accord.</p><div class="ref-gallery">${gallery}</div><p>Ces images aident à discuter d’une direction visuelle. Elles ne représentent aucune intervention ni aucun client de l’entreprise.</p></div></section><section class="ref-section"><div class="ref-wrap"><span class="ref-kicker">Un parcours simple</span><h2>Comment présenter un projet</h2><div class="ref-services"><article class="ref-service"><span class="ref-num">01</span><h3>Décrire le besoin</h3><p>Présenter le lieu, l’ouvrage et les attentes en quelques mots. Les dimensions, les accès et la localisation aident à préparer une première conversation.</p></article><article class="ref-service"><span class="ref-num">02</span><h3>Échanger directement</h3><p>Contacter l’entreprise au numéro professionnel publié pour connaître ses disponibilités. Le premier échange permet aussi de vérifier le périmètre du métier et la zone d’intervention.</p></article><article class="ref-service"><span class="ref-num">03</span><h3>Préciser la suite</h3><p>Définir avec elle les modalités d’intervention et de devis. Cette maquette ne fixe ni tarif, ni calendrier, ni méthode commerciale à sa place.</p></article></div></div></section><section class="ref-section ref-section--soft" id="communes"><div class="ref-wrap ref-two"><div><span class="ref-kicker">À proximité</span><h2>Des pages par commune</h2><p>Ces pages locales sont des propositions. Elles ne confirment pas que ${esc(name)} intervient dans chacune de ces communes.</p></div><div class="ref-local-links">${communeLinks()}</div></div></section><section class="ref-section"><div class="ref-wrap ref-two"><div><span class="ref-kicker">Questions utiles</span><h2>Avant de se lancer</h2></div><div class="ref-faq"><h3>Ces images montrent-elles des chantiers de ${esc(name)} ?</h3><p>Non. Ce sont des photos de démonstration issues de la bibliothèque Presti. Leurs auteurs et licences sont cités. Les images propres à l’entreprise ne seront intégrées qu’avec son accord.</p><h3>Les prestations affichées sont-elles confirmées ?</h3><p>Non. Elles forment une proposition de navigation adaptée au métier déclaré. L’entreprise pourra conserver, corriger ou retirer chaque rubrique pendant l’échange.</p><h3>Peut-on demander un devis depuis cette maquette ?</h3><p>Le téléphone permet un contact direct. Aucun formulaire de devis n’est actif ici et aucune demande en ligne n’est transmise.</p><h3>Ce site est-il publié ?</h3><p>Non. Il s’agit d’un exemple privé destiné à préparer une conversation. La mise en ligne et les textes définitifs exigeraient une validation préalable.</p></div></div></section>${contact}<section class="ref-section"><div class="ref-wrap ref-two"><div><span class="ref-kicker">À construire ensemble</span><h2>Une page fidèle au terrain</h2></div><div><p>Les mots de cette maquette ne remplacent pas la parole de l’artisan. Lors d’un échange, chaque intitulé peut être reformulé, les exemples trop larges peuvent être retirés et les informations utiles aux clients peuvent être ajoutées.</p><p>Avant toute diffusion, il faudra confirmer les prestations, la zone couverte, les coordonnées et les modalités de réponse. Cette étape distingue la proposition visuelle de la présentation officielle de l’entreprise.</p></div></div></section>`
 const prestationPage=site.pages.find(p=>p.type==='prestations'), contactPage=site.pages.find(p=>p.type==='contact'),legalPage=site.pages.find(p=>p.type==='legal')
 const imgSide=rest[0]||hero
 const prestBody=`<section class="ref-pagehero"><div class="ref-wrap"><span class="ref-kicker">${esc(label)} · ${esc(city)}</span><h1>Les prestations possibles</h1><p>Une proposition de rubriques à discuter avec ${esc(name)}. Aucune prestation précise n’est présentée comme confirmée.</p></div></section><section class="ref-section"><div class="ref-wrap ref-two"><div><span class="ref-kicker">Une structure à valider</span><h2>${esc(content.pages['/prestations/'].sections[0].h2)}</h2><p>${esc(intro)}</p><p>Les intitulés et les textes ci-dessous reprennent la première proposition. L’entreprise pourra les corriger avant toute diffusion.</p><div class="ref-disclaimer">Maquette — exemple non contractuel. Aucun chantier présenté ici n’est attribué à ${esc(name)}.</div></div><figure>${image(imgSide,'../')}<figcaption>${credit(imgSide)}</figcaption></figure></div><div class="ref-wrap ref-services ref-services--stack">${cardServices}</div></section><section class="ref-contact"><div class="ref-wrap"><h2>Une demande précise ?</h2><p>Pour connaître les prestations effectivement proposées, appelez directement l’entreprise.</p><a class="ref-phone" href="${tel(phone)}">${esc(phone)}</a></div></section>`
 const contactBody=`<section class="ref-pagehero"><div class="ref-wrap"><span class="ref-kicker">Contact · ${esc(city)}</span><h1>Entrons en contact</h1><p>Un seul geste : joindre l’entreprise par son numéro professionnel publié.</p></div></section><section class="ref-section"><div class="ref-wrap ref-two"><div><span class="ref-kicker">${esc(name)}</span><h2>Parlons du projet</h2><p>Décrivez votre besoin, l’adresse de l’intervention et le calendrier souhaité. L’entreprise vous indiquera elle-même si elle peut y répondre.</p><a class="ref-phone ref-phone--dark" href="${tel(phone)}">${esc(phone)}</a><p>Numéro issu d’un annuaire ou d’une fiche professionnelle publique. À reconfirmer au premier appel.</p><div class="ref-disclaimer">Cette maquette ne comporte pas de formulaire. Aucun message n’est collecté ni envoyé.</div></div><figure>${image(imgSide,'../')}<figcaption>${credit(imgSide)}</figcaption></figure></div></section>`
 const credits=attrs.map(a=>`<li><a href="${esc(a.source)}" rel="noopener" target="_blank">${esc(a.titre_commons||a.legende)}</a> — ${esc(a.auteur)} · <a href="${licenceUrl(a.licence)}" rel="noopener" target="_blank">${esc(a.licence)}</a> · illustration de démonstration${a.modifiee?' · image modifiée':''}</li>`).join('')
 const legalBody=`<section class="ref-pagehero"><div class="ref-wrap"><span class="ref-kicker">Document privé</span><h1>Mentions de la maquette</h1><p>Exemple non contractuel, préparé pour discussion. Aucun site n’est publié.</p></div></section><section class="ref-section"><div class="ref-wrap"><h2>Sources et images</h2><p>Entreprise : ${esc(name)} · SIREN ${esc(site.siren)} · ${esc(city)}. Téléphone professionnel : <a href="${tel(phone)}">${esc(phone)}</a>.</p><p>Source du numéro : ${esc(site.source_telephone)}. Ce numéro reste à confirmer directement avec l’entreprise.</p><p>Les photographies sont des illustrations du métier provenant des sites de démonstration Presti. Elles ne représentent pas les réalisations de l’entreprise. Aucune photo de réseau social n’a été copiée.</p><ul class="ref-credits">${credits}</ul><p>Les auteurs, licences et pages d’origine sont repris des attributions conservées dans le dossier privé.</p></div></section>`
 const map=new Map([[prestationPage.chemin,prestBody],[contactPage.chemin,contactBody],[legalPage.chemin,legalBody],['/',homeBody]])
 for(const page of site.pages.filter(p=>p.type==='commune')){
  const c=content.pages[page.chemin],near=site.communes.slice(1).filter(x=>x!==page.cible)
  map.set(page.chemin,`<section class="ref-hero ref-hero--${art.hero} ref-hero--local">${image(hero,'../','ref-hero__img')}<div class="ref-wrap ref-hero__in"><span class="ref-kicker">${esc(label)} · ${esc(page.cible)}</span><h1>${esc(c.h1)}</h1><p>Page locale de maquette pour ${esc(name)}. La zone d’intervention reste à confirmer directement avec l’entreprise.</p><a class="ref-button" href="../contact/">Contacter l’entreprise ↗</a></div><span class="ref-hero__credit">${credit(hero)}</span></section><section class="ref-section"><div class="ref-wrap ref-two"><div><span class="ref-kicker">Repère local</span><h2>Le métier, la commune</h2></div><div><p>${esc(c.sections[0].blocs[0].texte)}</p><p>Les photographies illustrent le métier. Elles ne montrent pas de chantier de ${esc(name)} à ${esc(page.cible)}.</p><a class="ref-phone ref-phone--dark" href="${tel(phone)}">${esc(phone)}</a><p>Pour vérifier les prestations et la zone desservie, appelez le numéro professionnel publié.</p></div></div></section><section class="ref-section ref-section--soft"><div class="ref-wrap"><h2>Communes proches</h2><div class="ref-local-links">${near.map(x=>`<a href="../${pageFor(x).chemin.slice(1)}">${esc(x)} ↗</a>`).join('')}</div></div></section>`)
 }
 const fontNames=[...new Set([art.display,art.texte])]
 const fonts=path.join(out,'fonts');fs.mkdirSync(fonts,{recursive:true});for(const f of fontNames){const file=`${slug(f)}.woff2`;fs.copyFileSync(path.join(here,'_polices',slug(f),file),path.join(fonts,file));fs.copyFileSync(path.join(here,'_polices',slug(f),'OFL.txt'),path.join(fonts,`${slug(f)}-OFL.txt`))}
 const faces=fontNames.map(f=>`@font-face{font-family:'${f}';src:url('fonts/${slug(f)}.woff2') format('woff2');font-weight:100 900;font-display:swap}`).join('\n')
 fs.writeFileSync(path.join(out,'kit.css'),fs.readFileSync(path.join(here,'_kit/kit.css')))
 fs.writeFileSync(path.join(out,'kit.js'),fs.readFileSync(path.join(here,'_kit/kit.js')))
 fs.writeFileSync(path.join(out,'styles.css'),`${faces}\n:root{--accent:${art.accent};--accent-texte:${art.accentTexte};--fond:${art.fond};--fond-2:${art.fond2};--encre:${art.encre};--encre-2:${art.encre2};--ligne:${art.ligne};--police-display:'${art.display}',serif;--police-texte:'${art.texte}',sans-serif;--texture:${art.texture};}\n${fs.readFileSync(path.join(here,'references.css'))}`)
 for(const page of site.pages){const file=path.join(out,page.chemin==='/'?'index.html':page.chemin.slice(1)+'index.html');fs.mkdirSync(path.dirname(file),{recursive:true});fs.writeFileSync(file,shell(page,map.get(page.chemin)))}
 fs.writeFileSync(path.join(out,'robots.txt'),'User-agent: *\nDisallow: /\n')
 console.log(`${dir}: ${site.pages.length} pages, ${art.display}/${art.texte}, héros ${art.hero}`)
}
const dossierCible=process.argv.find(x=>x.startsWith('--dossier='))?.slice('--dossier='.length)
for(const d of fs.readdirSync(root).filter(x=>/^\d\d-/.test(x) && (!dossierCible||x===dossierCible)))render(d)
