def esc(s): return html.escape(str(s), quote=True).replace('&#x27;', "'")
def slug(s):
 import unicodedata
 s = unicodedata.normalize('NFD', s)
 s = ''.join(c for c in s if unicodedata.category(c) != 'Mn').lower()
 return re.sub(r'(^-|-$)', '', re.sub('[^a-z0-9]+', '-', s))
def fill(v, name, city):
 if isinstance(v, str): return v.replace('{entreprise}', name).replace('{ville}', city)
 if isinstance(v, list): return [fill(x, name, city) for x in v]
 if isinstance(v, dict): return {k: fill(x, name, city) for k, x in v.items()}
 return v

def photo_set(job):
 folder = DEMOS / job
 attrs = json.loads((folder / 'attributions.json').read_text())
 files = {x['fichier'].split('/')[-1]:x for x in attrs}
 names = ['hero-1600.webp']
 for n in sorted(files):
  if n.endswith('-960.webp') and not n.startswith('hero-') and n not in names:
   names.append(n)
   if len(names) == 5: break
 if len(names) < 4: raise RuntimeError(f'Photos insuffisantes : {job}')
 return names, [files[n] for n in names]
