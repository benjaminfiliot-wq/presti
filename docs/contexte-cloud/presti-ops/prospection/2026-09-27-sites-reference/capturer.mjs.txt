import { chromium } from '/home/usine/presti/node_modules/playwright/index.mjs'
import { readdirSync } from 'node:fs'
import { join } from 'node:path'
const root = '/home/usine/veille/prospection/sites-reference'
const cible = process.argv.find(x => x.startsWith('--dossier='))?.slice('--dossier='.length)
const dirs = readdirSync(root, { withFileTypes: true }).filter(d => d.isDirectory() && /^\d\d-/.test(d.name) && (!cible || d.name === cible)).map(d => d.name).sort()
const browser = await chromium.launch({ headless: true, executablePath: '/home/usine/.cache/ms-playwright/chromium_headless_shell-1243/chrome-headless-shell-linux-arm64/chrome-headless-shell' })
try {
  for (const dir of dirs) {
    for (const [kind, width, height] of [['bureau',1440,900],['telephone',390,844]]) {
      const page = await browser.newPage({ viewport: { width, height }, deviceScaleFactor: 1, reducedMotion: 'no-preference' })
      const url = `http://127.0.0.1:8765/${dir}/site/`
      await page.goto(url, { waitUntil: 'networkidle' })
      await page.evaluate(() => document.fonts.ready)
      await page.waitForTimeout(1600)
      const state = await page.evaluate(() => ({ title: document.title, overflow: document.documentElement.scrollWidth > innerWidth, imagesBroken: Array.from(document.images).filter(x => x.complete && !x.naturalWidth).map(x => x.src) }))
      await page.screenshot({ path: join(root,dir,`capture-${kind}.png`), fullPage: false, animations: 'disabled' })
      console.log(dir,kind,JSON.stringify(state))
      await page.close()
    }
  }
} finally { await browser.close() }
