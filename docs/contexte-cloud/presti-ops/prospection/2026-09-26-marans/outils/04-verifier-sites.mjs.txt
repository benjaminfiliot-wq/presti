// Étape 4 : pour chaque entreprise ayant un site_internet connu (champ ADEME), on fait une requête
// HTTP légère (une page, timeout 10s) pour juger : répond ou non, HTTPS, balise viewport, année copyright, formulaire contact.
// Classement : A = pas de site connu | B = site KO / pas mobile / copyright <= 2021 | C = site correct.
import fs from "node:fs";

const IN = new URL("../data/03-entreprises-actives.json", import.meta.url);
const OUT = new URL("../data/04-entreprises-classees.json", import.meta.url);

const entreprises = JSON.parse(fs.readFileSync(IN, "utf8"));

function normalizeUrl(u) {
  if (!u) return null;
  let s = u.trim();
  if (!s) return null;
  if (!/^https?:\/\//i.test(s)) s = "https://" + s;
  try { new URL(s); return s; } catch { return null; }
}

async function checkSite(rawUrl) {
  const url = normalizeUrl(rawUrl);
  if (!url) return { constat: "site indiqué mais URL invalide", classe: "B", url_testee: rawUrl };

  const controller = new AbortController();
  const timeout = setTimeout(() => controller.abort(), 10000);
  try {
    let res = await fetch(url, { signal: controller.signal, redirect: "follow" });
    clearTimeout(timeout);
    if (!res.ok) {
      return { constat: `répond avec HTTP ${res.status}`, classe: "B", url_testee: url, https: url.startsWith("https") };
    }
    const finalUrl = res.url || url;
    const https = finalUrl.startsWith("https://");
    let html = "";
    try { html = await res.text(); } catch { html = ""; }
    const lower = html.toLowerCase();
    const hasViewport = /<meta[^>]+name=["']viewport["']/.test(lower);
    const yearMatches = [...lower.matchAll(/(20[0-2][0-9])/g)].map((m) => parseInt(m[1], 10));
    const copyrightYear = yearMatches.length ? Math.max(...yearMatches) : null;
    const hasContactForm = /<form/.test(lower) && (/contact/.test(lower) || /formulaire/.test(lower));

    const problems = [];
    if (!https) problems.push("pas de HTTPS");
    if (!hasViewport) problems.push("pas de balise viewport (non mobile)");
    if (copyrightYear && copyrightYear <= 2021) problems.push(`copyright ${copyrightYear}`);

    const classe = problems.length ? "B" : "C";
    const constat = problems.length
      ? `site répond mais : ${problems.join(", ")}`
      : `site correct (HTTPS, mobile${copyrightYear ? `, copyright ${copyrightYear}` : ""}${hasContactForm ? ", formulaire de contact" : ""})`;

    return { constat, classe, url_testee: finalUrl, https, viewport: hasViewport, copyright_annee: copyrightYear, formulaire_contact: hasContactForm };
  } catch (err) {
    clearTimeout(timeout);
    const msg = err.name === "AbortError" ? "ne répond pas (timeout 10s)" : `ne répond pas (${err.code || err.message || "erreur réseau"})`;
    return { constat: msg, classe: "B", url_testee: url };
  }
}

const out = [];
let i = 0;
for (const e of entreprises) {
  i++;
  process.stdout.write(`\r${i}/${entreprises.length}`);
  if (!e.site_internet || !e.site_internet.trim()) {
    out.push({ ...e, priorite: "A", constat_site: "aucun site connu (absent des sources ADEME/API)" });
    continue;
  }
  const r = await checkSite(e.site_internet);
  out.push({ ...e, priorite: r.classe, constat_site: r.constat, site_internet: r.url_testee || e.site_internet });
}
console.log("");

const counts = { A: 0, B: 0, C: 0 };
for (const e of out) counts[e.priorite]++;
console.log("Répartition A/B/C :", counts);

fs.writeFileSync(OUT, JSON.stringify(out, null, 2));
console.log(`Écrit -> data/04-entreprises-classees.json (${out.length} entreprises)`);
