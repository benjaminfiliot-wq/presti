// Dossier de profil unique à cette exécution (un essai précédent, tué par le timeout, peut avoir
// laissé un verrou de profil qui bloquerait un nouveau Chrome pointant sur le même dossier).
const PROFILE_DIR = path.join(process.env.TEMP || process.env.TMP || ".", `presti-ops-chrome-profile-marans-${Date.now()}`);
function screenshot(url, outPng, w, h) {
  // Flags minimaux, ceux vérifiés à la main en ligne de commande (fonctionnent en ~2 s) —
  // les flags supplémentaires essayés avant (virtual-time-budget, disable-background-networking...)
  // faisaient tourner Chrome en boucle sans jamais rendre. timeout : filet de sécurité.
  execFileSync(CHROME, [
    "--headless=new", "--disable-gpu", "--no-sandbox",
    `--user-data-dir=${PROFILE_DIR}`,
    `--window-size=${w},${h}`, `--screenshot=${outPng}`, url,
  ], { stdio: "pipe", timeout: 15000 });
}

// IMPORTANT : le serveur DOIT être un process séparé, pas celui-ci. execFileSync (screenshot)
// bloque tout le thread Node le temps que Chrome tourne ; un http.createServer() dans CE process
// ne peut alors plus répondre aux requêtes de Chrome pendant l'attente -> blocage croisé observé
// (Chrome attend une réponse que ce même process, bloqué dans spawnSync, ne peut pas envoyer).
// D'où : lancer `python -m http.server <port>` à part AVANT ce script, et pointer PORT dessus.
const PORT = Number(process.env.PORT_SERVEUR || 8793);
console.log(`Utilise le serveur local déjà lancé sur http://127.0.0.1:${PORT} (loopback uniquement).`);

for (const d of dossiers) {
  const dir = path.join(MAQUETTES, d.dossier);
  const genDir = path.join(dir, "_gen");
  fs.mkdirSync(genDir, { recursive: true });

  const badge = `${contenuIso.libelles.sur.toUpperCase()} · ${d.ville.toUpperCase()}`;
  const h1 = `${contenuIso.libelles.sur} à ${d.ville}`;
  const sub = contenuIso.pages["/"].h1Ligne;
  const chapeau = contenuIso.pages["/"].chapeau;
  const photoUrlBureau = `/maquettes/${d.dossier}/capture-bureau.jpg`;
  const photoUrlTel = `/maquettes/${d.dossier}/capture-telephone.jpg`;

  const bureauHtmlPath = path.join(genDir, "bureau.html");
  const telHtmlPath = path.join(genDir, "telephone.html");
  fs.writeFileSync(bureauHtmlPath, htmlBureau({ entreprise: d.entreprise, ville: d.ville, badge, h1, sub, chapeau, photoUrl: photoUrlBureau }));
  fs.writeFileSync(telHtmlPath, htmlTelephone({ entreprise: d.entreprise, ville: d.ville, badge, h1, sub, chapeau, photoUrl: photoUrlTel, navH: NAV_MOBILE[d.dossier] }));

  const relB = path.relative(ROOT, bureauHtmlPath).replace(/\\/g, "/");
  const relT = path.relative(ROOT, telHtmlPath).replace(/\\/g, "/");
  const pngB = path.join(genDir, "bureau.png");
  const pngT = path.join(genDir, "telephone.png");

  console.log(`== ${d.dossier} ==`);
  screenshot(`http://127.0.0.1:${PORT}/${relB}`, pngB, 1440, 900);
  screenshot(`http://127.0.0.1:${PORT}/${relT}`, pngT, 390, 844);

  const outB = path.join(dir, "capture-bureau.jpg");
  const outT = path.join(dir, "capture-telephone.jpg");
  execFileSync("ffmpeg", ["-y", "-i", pngB, "-q:v", "3", outB], { stdio: "pipe" });
  execFileSync("ffmpeg", ["-y", "-i", pngT, "-q:v", "3", outT], { stdio: "pipe" });
  console.log(`  -> ${outB}`);
  console.log(`  -> ${outT}`);

  fs.rmSync(genDir, { recursive: true, force: true });
}

fs.rmSync(PROFILE_DIR, { recursive: true, force: true });
console.log("Terminé.");
