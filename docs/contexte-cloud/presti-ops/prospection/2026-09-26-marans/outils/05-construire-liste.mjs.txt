// Étape 5 : assemble la liste finale (CSV) + le résumé (Markdown).
// Cap visé : 60 à 120 entreprises, en gardant TOUTES les A et B (meilleures cibles),
// et en complétant avec quelques C seulement si besoin pour rester dans la fourchette.
import fs from "node:fs";

const IN = new URL("../data/04-entreprises-classees.json", import.meta.url);
const CSV_OUT = new URL("../liste.csv", import.meta.url);
const MD_OUT = new URL("../liste-resume.md", import.meta.url);

const all = JSON.parse(fs.readFileSync(IN, "utf8"));

const A = all.filter((e) => e.priorite === "A").sort((a, b) => a.commune.localeCompare(b.commune) || a.nom_entreprise.localeCompare(b.nom_entreprise));
const B = all.filter((e) => e.priorite === "B").sort((a, b) => a.commune.localeCompare(b.commune) || a.nom_entreprise.localeCompare(b.nom_entreprise));
const C = all.filter((e) => e.priorite === "C").sort((a, b) => a.commune.localeCompare(b.commune) || a.nom_entreprise.localeCompare(b.nom_entreprise));

const CAP = 120;
let finale = [...A, ...B];
if (finale.length < 60) {
  // complète avec des C si jamais A+B ne suffisait pas (pas le cas ici mais on reste robuste)
  finale = [...finale, ...C.slice(0, Math.max(0, 60 - finale.length))];
} else if (finale.length > CAP) {
  finale = finale.slice(0, CAP);
} else {
  // on ajoute quelques C (fin de liste) tant qu'on ne dépasse pas le cap
  const room = CAP - finale.length;
  finale = [...finale, ...C.slice(0, room)];
}

console.log(`Liste finale : ${finale.length} entreprises (A=${A.length}, B=${B.length}, C retenues=${finale.length - A.length - B.length}/${C.length})`);

// --- CSV ---
function csvEscape(v) {
  if (v === null || v === undefined) return "";
  const s = String(v);
  if (/[",;\n]/.test(s)) return `"${s.replace(/"/g, '""')}"`;
  return s;
}
const SOURCE = "ADEME RGE (data.ademe.fr, dataset liste-des-entreprises-rge-2) + recherche-entreprises.api.gouv.fr";
const header = ["priorite", "entreprise", "metier", "commune", "telephone", "site", "constat_site", "siret", "date_creation", "effectif", "source"];
const lines = [header.join(";")];
for (const e of finale) {
  lines.push([
    e.priorite,
    e.nom_entreprise,
    e.metiers,
    e.commune,
    e.telephone,
    e.site_internet || "",
    e.constat_site,
    e.siret,
    e.date_creation,
    e.effectif,
    SOURCE,
  ].map(csvEscape).join(";"));
}
fs.writeFileSync(CSV_OUT, lines.join("\n"), "utf8");
console.log(`Écrit -> liste.csv (${finale.length} lignes)`);

// --- Résumé Markdown ---
function tally(arr, key) {
  const m = {};
  for (const e of arr) m[e[key]] = (m[e[key]] || 0) + 1;
  return Object.entries(m).sort((a, b) => b[1] - a[1]);
}
function tallyMetier(arr) {
  const m = {};
  for (const e of arr) for (const met of e.metiers.split(" + ")) m[met] = (m[met] || 0) + 1;
  return Object.entries(m).sort((a, b) => b[1] - a[1]);
}

const parCommune = tally(finale, "commune");
const parMetier = tallyMetier(finale);
const parPriorite = tally(finale, "priorite");

const md = `# Prospection artisans RGE — autour de Marans (17230)
Liste construite le 2026-09-25 pour les appels du samedi 26/09 au soir.

## Sources (gratuites, publiques, citées)
- **ADEME — liste des entreprises RGE** (open data) : dataset \`liste-des-entreprises-rge-2\`, API \`https://data.ademe.fr/data-fair/api/v1/datasets/liste-des-entreprises-rge-2/lines\` (filtre \`qs=code_postal:XXXXX\`). Page dataset : https://data.ademe.fr/datasets/liste-des-entreprises-rge-2
- **API Recherche d'entreprises** (data.gouv.fr / DINUM, gratuite) : \`https://recherche-entreprises.api.gouv.fr/search?q=<SIRET>\` — utilisée pour l'état administratif, la date de création et l'effectif. Doc : https://recherche-entreprises.api.gouv.fr/docs/

## Zone couverte (~35 km autour de Marans) — codes postaux utilisés
17230 (Marans, Andilly, Charron, Villedoux, Longèves, Saint-Ouen-d'Aunis), 17137 (Esnandes, Nieul-sur-Mer, Marsilly, L'Houmeau), 17540 (Nuaillé-d'Aunis), 17170 (Courçon, Saint-Jean-de-Liversay, Ferrières, La Ronde, Taugon), 17139 (Dompierre-sur-Mer), 17138 (Puilboreau, Saint-Xandre), 17180 (Périgny), 17000 (La Rochelle), 17140 (Lagord), 17440 (Aytré), 17700 (Surgères), 85450 (Chaillé-les-Marais), 85400 (Luçon), 85770 (L'Île-d'Elle).

## Méthode de filtrage
1. Téléchargement de toutes les lignes RGE pour ces 14 codes postaux (558 lignes brutes, une ligne = une qualification).
2. Filtre sur le domaine de qualification RGE, rattaché à un métier : plomberie-chauffage (chaudière bois/condensation, PAC, poêle, solaire thermique, VMC), électricité (photovoltaïque, radiateurs électriques), menuiserie (fenêtres/volets/portes), couverture (isolation toiture/terrasse), et isolation (proche, combles/murs/planchers) → 447 lignes.
3. Regroupement par SIRET (une entreprise peut avoir plusieurs qualifications) → 180 entreprises uniques, 179 avec un téléphone public.
4. Vérification de l'état administratif via l'API Recherche d'entreprises → 177 entreprises actives (2 fermées écartées).
5. Vérification du site web connu (champ \`site_internet\` de l'ADEME) par requête HTTP légère (une page, délai 10 s) : répond ou non, HTTPS, balise viewport, année de copyright, formulaire de contact.
   - **A = aucun site connu** (rien dans les sources ADEME/API) → meilleure cible : ${A.length}
   - **B = site connu mais problématique** (ne répond pas, pas de HTTPS, pas de balise viewport mobile, ou copyright ≤ 2021) : ${B.length}
   - **C = site correct** (répond, HTTPS, mobile, récent) → gardé en fin de liste : ${C.length} au total, ${finale.length - A.length - B.length} retenues pour compléter la liste

⚠️ Limite honnête : le site web n'est vérifié que s'il est déclaré dans la fiche RGE de l'ADEME. Une entreprise classée A peut en réalité avoir un site non déclaré à l'ADEME — aucune recherche web individuelle par entreprise n'a été faite (uniquement des sources API structurées, sans passer les données de prospects à un moteur de recherche ou une IA).

## Liste finale retenue : ${finale.length} entreprises (visée 60–120, au moins 40 en A+B)

### Par priorité
${parPriorite.map(([k, v]) => `- ${k} : ${v}`).join("\n")}

### Par métier
${parMetier.map(([k, v]) => `- ${k} : ${v}`).join("\n")}

### Par commune
${parCommune.map(([k, v]) => `- ${k} : ${v}`).join("\n")}

## Fichier
\`liste.csv\` — colonnes : priorite, entreprise, metier, commune, telephone, site, constat_site, siret, date_creation, effectif, source. Trié A puis B puis C.

## Rappel avant d'appeler
- Ce fichier reste en local, hors de tout dépôt Git, jamais commité.
- Aucun message n'a été envoyé à un prospect : la liste ne fait que préparer les appels de Benj.
`;

fs.writeFileSync(MD_OUT, md, "utf8");
console.log(`Écrit -> liste-resume.md`);
